  filter(_eq?, _in?) {
    const filters = {};
    for (const key in _eq) {
      filters[key] = {op: 'eq', unary: _eq[key]};
    }
    for (const key in _in) {
      filters[key] = {op: 'in', unary: _in[key]};
    }
    return {filter: [filters]};
  }
