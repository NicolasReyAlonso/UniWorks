import {Injectable} from '@angular/core';
import {environment} from '../../environments/environment';

@Injectable({
  providedIn: 'root'
})
export class GlobalService {

  // Configuration variables
  parameters = {
    // tslint:disable-next-line:max-line-length
    host: environment.backend_url, // 'http://localhost:5000', 'http://localhost:5000', 'http://10.141.187.173:5000', 'https://one.nis.magic-nexus.eu'
    api_prefix: '/api',
  };

  authOptions: any; // Used in every REST call

  constructor() {
  }

  getParameter(k: string): any {
    return this.parameters[k];
  }

  setHostUrl(url): void {
    this.parameters.host = url;
  }

  isInt(n) {
    return Number(n) === n && n % 1 === 0;
  }

  isFloat(n) {
    return Number(n) === n && n % 1 !== 0;
  }

  isNumber(n) {
    return Number(n) === n;
  }

  convertToJson(jsonString: string): any {
    try {
      return JSON.parse(jsonString);
    } catch (e) {
      return null;
    }
  }
}
