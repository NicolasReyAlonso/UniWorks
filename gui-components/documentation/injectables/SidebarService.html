import { EventEmitter, Inject, Injectable, Optional } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import * as yaml from 'js-yaml';
import { SidebarItem } from '@interfaces/sidebar-item.interfaces';
import { AuthService } from '@services/auth.service';
import { firstValueFrom } from 'rxjs';
import { SIDEBAR_YAML_PATH } from '@tokens/sidebar-config.token';

/**
 * @service SidebarService
 * @description
 * Servicio encargado de cargar y procesar la configuración del sidebar desde un archivo YAML.
 * También expone un `EventEmitter` para comunicar eventos relacionados con el sidebar.
 */
@Injectable({
  providedIn: 'root'
})
export class SidebarService {

  /** Emisor de eventos personalizados del sidebar (ej. selección de ruta manual) */
  sidebarEventEmitter = new EventEmitter<{ type: string, value: any }>();

  /** Ruta por defecto del archivo YAML del sidebar */
  private defaultYamlPath = '/assets/config/submenu.yaml';

  /**
   * Constructor del servicio Sidebar
   * @param http Cliente HTTP de Angular para cargar el YAML
   * @param authService Servicio de autenticación y permisos
   * @param yamlPath (Opcional) Token de inyección que permite sobrescribir la ruta del YAML
   */
  constructor(
    private http: HttpClient,
    private authService: AuthService,
    @Optional() @Inject(SIDEBAR_YAML_PATH) private yamlPath: string
  ) {}

  /**
   * Carga un archivo YAML desde la ruta especificada y lo convierte en un objeto `SidebarItem`.
   * Además, inyecta dinámicamente funciones de condición (`funcCondition`) según los permisos definidos.
   *
   * @param yamlPath Ruta del archivo YAML (por ejemplo: '/assets/config/sidebar.yaml')
   * @returns Promesa que resuelve con la estructura `SidebarItem` cargada
   * @throws Error si el archivo no se puede leer o parsear
   */
  async loadSidebarItems(yamlPath?: string): Promise<SidebarItem> {
    yamlPath = this.yamlPath || yamlPath || this.defaultYamlPath;

    try {
      // Leer el YAML como texto
      const yamlText = await firstValueFrom(
        this.http.get(yamlPath, { responseType: 'text' })
      );

      // Parsear el YAML a objeto JavaScript
      const parsed = yaml.load(yamlText) as SidebarItem;

      // Añadir dinámicamente funcCondition basado en "permission"
      Object.keys(parsed).forEach(submenuKey => {
        const submenu = parsed[submenuKey];
        Object.keys(submenu.routes).forEach(routeKey => {
          const route = submenu.routes[routeKey];
          if (route.permission) {
            route.funcCondition = () =>
              this.authService.havePermission(route.permission!);
          }
        });
      });

      return parsed;

    } catch (error) {
      console.error('Error cargando el sidebar YAML:', error);
      throw error;
    }
  }
}
