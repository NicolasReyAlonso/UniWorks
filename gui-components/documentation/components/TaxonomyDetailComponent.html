import { Component, OnInit } from '@angular/core';
import {BackendService} from '@services/backend.service';
import {Router} from '@angular/router';
import { CommonModule } from '@angular/common';
import {BOCardComponent} from "@components/miscellaneous/bo-card/bo-card.component";
import {NzTableModule} from "ng-zorro-antd/table";
import {SharedModule} from "../../../../shared-module/shared.module";
import {NzTabsModule} from "ng-zorro-antd/tabs";
import {NzPageHeaderModule} from "ng-zorro-antd/page-header";

@Component({
    selector: 'app-taxonomy-detail',
    imports: [
        CommonModule,
        BOCardComponent,
        NzTableModule,
        SharedModule,
        NzTabsModule,
        NzPageHeaderModule,
    ],
    templateUrl: './taxonomy-detail.component.html',
    styleUrls: ['./taxonomy-detail.component.sass']
})
export class TaxonomyDetailComponent implements OnInit {

  constructor(private backendService: BackendService,
    private router: Router) { }

  taxonomy_id: string;
  taxonomy: any = {};
  dbxref: any = {};
  organisms: any = [];
  expandSet = new Set<number>();

  ngOnInit(): void {
    this.taxonomy_id = this.router.url.replace(/^.*[\\\/]/, '');
    this.backendService.getTaxonomies(this.taxonomy_id).subscribe(
      {next: response => this.taxonomy=response['content']});
    this.backendService.getOrganisms(undefined, this.backendService.filter({'phylotree_id':this.taxonomy_id})).subscribe(
          {next: response => this.organisms=response['content']});
  }

  onExpandChange(id: number): void {
    if (!this.expandSet.has(id)) {
      this.expandSet.add(id);
    } else {
      this.expandSet.delete(id);
    }
  }

}
