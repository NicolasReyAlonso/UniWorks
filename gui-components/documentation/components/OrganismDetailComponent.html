import {Component, OnDestroy, OnInit} from '@angular/core';
import {BackendService} from '../../../../services/backend.service';
import {ActivatedRoute, Router} from '@angular/router';
import IView from '../../../../interfaces/view.interfaces';
import {FormsModule, ReactiveFormsModule, UntypedFormBuilder, UntypedFormGroup} from '@angular/forms';
import {StateService} from '../../../../services/state.service';
import {AuthService} from '../../../../services/auth.service';
import {NotificationService, TypeNotificationEnum} from '../../../../services/notification.service';
import {MessageLogService} from '../../../../services/message-log.service';
import OBJECT_TYPES from '../../../../constants/object-types.constants';
import { CommonModule } from '@angular/common';
import {SharedModule} from "../../../../shared-module/shared.module";
import {NzTableModule} from "ng-zorro-antd/table";
import {NzTabsModule} from "ng-zorro-antd/tabs";
import {NzInputModule} from "ng-zorro-antd/input";
import {NzSpinModule} from "ng-zorro-antd/spin";
import {DisableControlDirective} from "../../../../directives/disable-control.directive";
import {NzFormModule} from "ng-zorro-antd/form";
import {NzPageHeaderModule} from "ng-zorro-antd/page-header";
import {BackButtonComponent} from "../../../../components/miscellaneous/back-button/back-button.component";
import {NzButtonModule} from "ng-zorro-antd/button";
import {lastValueFrom} from "rxjs";

@Component({
    selector: 'app-organism-detail',
    imports: [
        CommonModule,
        SharedModule,
        NzTableModule,
        NzTabsModule,
        NzInputModule,
        NzSpinModule,
        DisableControlDirective,
        NzFormModule,
        ReactiveFormsModule,
        FormsModule,
        NzPageHeaderModule,
        BackButtonComponent,
        NzButtonModule,
    ],
    templateUrl: './organism-detail.component.html',
    styleUrls: ['./organism-detail.component.sass']
})
export class OrganismDetailComponent implements OnInit, OnDestroy, IView {

  BREADCRUMB_NAME = {
    key: 'METADATA.ORGANISMS.DETAIL.BREADCRUMB',
    params: {
      name: '',
    }
  };

  loading = false;
  formGroup: UntypedFormGroup;
  formGroupChange = false;

  organism: any = {};
  features = [];
  featuresIndex = 1;
  featuresPageSize = 10;
  featuresTotal = 0;
  stocks = [];
  stocksIndex = 1;
  stocksPageSize = 10;
  stocksTotal = 0;
  analyses = [];
  analysesIndex = 1;
  analysesPageSize = 10;
  analysesTotal = 0;

  constructor(
    private readonly backendService: BackendService,
    private readonly router: Router,
    private readonly activatedRoute: ActivatedRoute,
    private readonly fb: UntypedFormBuilder,
    private readonly stateService: StateService,
    public readonly authService: AuthService,
    private readonly notificationService: NotificationService,
    private readonly messageLogService: MessageLogService,
  ) {}

  async ngOnInit(): Promise<void> {
    this.loading = true;
    this.organism = this.activatedRoute.snapshot.data.organism;
    this.initFormGroup();
    const getState = await this.stateService.getStateCurrenView();
    await this.setFeaturesTable(this.featuresIndex, this.featuresPageSize);
    await this.setStocksTable(this.stocksIndex, this.stocksPageSize);
    await this.setAnalysesTable(this.stocksIndex, this.stocksPageSize);
    if (!getState || !getState.formGroupChange) {
      await this.loadDataByRequest(this.organism);
    } else {
      this.formGroup.get('common_name').setValue(getState.common_name ? getState.common_name : '');
      this.formGroup.get('abbreviation').setValue(getState.abbreviation ? getState.abbreviation : '');
      this.formGroup.get('genus').setValue(getState.genus ? getState.genus : '');
      this.formGroup.get('species').setValue(getState.species ? getState.species : '');
      this.formGroup.get('infraspecific_name').setValue(getState.infraspecific_name ? getState.infraspecific_name : '');
      this.formGroup.get('type_id').setValue(getState.type_id ? getState.type_id : '');
      this.formGroupChange = getState.formGroupChange;
    }
    this.formGroup.valueChanges.subscribe({next:(change) => this.formValueChange(change)});
    this.loading = false;
  }

  ngOnDestroy() {
    this.BREADCRUMB_NAME.params.name = this.formGroup.get('name').value;
    if (!this.formGroupChange) {
      this.stateService.setStateCurrentView({
        formGroupChange: this.formGroupChange,
      }).then();
    } else {
      this.stateService.setStateCurrentView({
        name: this.formGroup.get('name').value,
        common_name: this.formGroup.get('common_name').value,
        abbreviation: this.formGroup.get('abbreviation').value,
        genus: this.formGroup.get('genus').value,
        species: this.formGroup.get('species').value,
        infraspecific_name: this.formGroup.get('infraspecific_name').value,
        type_id: this.formGroup.get('type_id').value,
        formGroupChange: this.formGroupChange,
      }).then();
    }
  }

  initFormGroup(): void {
    this.formGroup = this.fb.group({
      organism_id: [{value: this.organism.organism_id, disabled: true}],
      name: [{value: this.organism.name, disabled: true}],
      common_name: [''],
      abbreviation: [''],
      genus: [''],
      species: [''],
      infraspecific_name: [''],
      type_id: [''],
    });
  }

  formValueChange(change): void {
    this.formGroupChange = true;
  }

  async loadDataByRequest(requestContent): Promise<void> {
    this.formGroup.get('common_name').setValue(requestContent.common_name ? requestContent.common_name : '');
    this.formGroup.get('abbreviation').setValue(requestContent.abbreviation ? requestContent.abbreviation : '');
    this.formGroup.get('genus').setValue(requestContent.genus ? requestContent.genus : '');
    this.formGroup.get('species').setValue(requestContent.species ? requestContent.species : '');
    this.formGroup.get('infraspecific_name').setValue(requestContent.infraspecific_name ? requestContent.infraspecific_name : '');
    this.formGroup.get('type_id').setValue(requestContent.type_id ? requestContent.type_id : '');
  }

  isDisableEdit(): boolean {
    return !this.authService.havePermission('gui-organism-edit');
  }

  async onPageIndexChangeFeatures(index): Promise<void> {
    this.loading = true;
    await this.setFeaturesTable(index, this.featuresPageSize);
    this.loading = false;
  }

  async setFeaturesTable(pageIndex, pageSize): Promise<void> {
    try {
      this.featuresIndex = pageIndex;
      const response: any = await lastValueFrom(this.backendService.getSequences(null, {
        filter: {organism_id: this.organism.organism_id},
        pagination: {
          pageIndex,
          pageSize,
        }
      }), { defaultValue: undefined });
      this.featuresTotal = response.count;
      this.features = response.content;
    } catch (e) {

    }
  }

  async onPageIndexChangeStocks(index): Promise<void> {
    this.loading = true;
    await this.setStocksTable(index, this.stocksPageSize);
    this.loading = false;
  }

  async setStocksTable(pageIndex, pageSize): Promise<void> {
    try {
      this.stocksIndex = pageIndex;
      const response: any = await lastValueFrom(this.backendService.getIndividuals(null, {
        organism_id: this.organism.organism_id,
        pagination: {
          pageIndex,
          pageSize,
        }
      }), { defaultValue: undefined });
      this.stocksTotal = response.count;
      this.stocks = response.content;
    } catch (e) {

    }
  }

  async onPageIndexChangeAnalyses(index): Promise<void> {
    this.loading = true;
    await this.setAnalysesTable(index, this.analysesPageSize);
    this.loading = false;
  }

  async setAnalysesTable(pageIndex, pageSize): Promise<void> {
    try {
      this.analysesIndex = pageIndex;
      const response: any = await lastValueFrom(this.backendService.getAnalyses(null, {
        filter: {organism_id: this.organism.organism_id},
        pagination: {
          pageIndex,
          pageSize,
        }
      }), { defaultValue: undefined });
      this.analysesTotal = response.count;
      this.analyses = response.content;
    } catch (e) {

    }
  }

  async saveChanges(): Promise<void> {
    try {
      this.loading = true;
      const data = {
        common_name: this.formGroup.get('common_name').value,
        abbreviation: this.formGroup.get('abbreviation').value,
        genus: this.formGroup.get('genus').value,
        species: this.formGroup.get('species').value,
        infraspecific_name: this.formGroup.get('infraspecific_name').value,
        type_id: this.formGroup.get('type_id').value,
      };
      const response: any = await lastValueFrom(this.backendService.putOrganisms(this.organism.organism_id, data), { defaultValue: undefined });
      this.messageLogService.addIssues(response.issues);
      this.formGroupChange = false;
    } catch (e) {
      if (e.issues) {
        this.messageLogService.addIssues(e.issues);
      }
      this.notificationService.createNotificationWithType(
        TypeNotificationEnum.error,
        'METADATA.ORGANISMS.DETAIL.CANCEL_CHANGES.NOTIFICATION_ERROR_SAVE_CHANGES.TITLE',
        'METADATA.ORGANISMS.DETAIL.CANCEL_CHANGES.NOTIFICATION_ERROR_SAVE_CHANGES.CONTENT',
        'bottomRight'
      ).then();
    }
    this.loading = false;
  }

  async cancelChanges(): Promise<void> {
    try {
      this.loading = true;
      const response: any = await lastValueFrom(this.backendService.getOrganisms(this.organism.organism_id), { defaultValue: undefined });
      this.messageLogService.addIssues(response.issues);
      const content = response.content;
      await this.loadDataByRequest(content);
      this.formGroupChange = false;
    } catch (e) {
      if (e.issues) {
        this.messageLogService.addIssues(e.issues);
      }
      this.notificationService.createNotificationWithType(
        TypeNotificationEnum.error,
        'METADATA.ORGANISMS.DETAIL.CANCEL_CHANGES.NOTIFICATION_ERROR_LOAD_DATA.TITLE',
        'METADATA.ORGANISMS.DETAIL.CANCEL_CHANGES.NOTIFICATION_ERROR_LOAD_DATA.CONTENT',
        'bottomRight'
      ).then();
    }
    this.loading = false;
  }

  goToIndividual(id) {
    this.router.navigate(['individualDetail', id]);
  }

  goToSequence(id) {
    this.router.navigate(['sequenceDetail', id]);
  }

  goOntologyDetail(): void {
    this.router.navigate(['termDetail', this.formGroup.get('type_id').value]);
  }

  goAnalysisDetail(a) {
    let url = '';
    if (a.object_type) {
      url = OBJECT_TYPES[a.object_type].pageDetail;
    }
    this.router.navigate([url ?? 'analysisDetail', a.analysis_id]);
  }

}
