import {AfterViewInit, Component, OnDestroy, OnInit, ViewChild} from '@angular/core';
import {NzModalModule, NzModalService} from 'ng-zorro-antd/modal';
import {UntypedFormGroup, UntypedFormControl, Validators, FormsModule, ReactiveFormsModule} from '@angular/forms';
import {BackendService} from 'src/app/services/backend.service';
import {MessageLogService} from 'src/app/services/message-log.service';
import {StateService} from 'src/app/services/state.service';
import {ActivatedRoute, Router} from '@angular/router';
import {InternationalizationService} from "ngt-gui";import {DynamicBrowseComponent} from "../../../../components/dynamics/dynamic-browse/dynamic-browse.component";
import IView from '@interfaces/view.interfaces';
import {AuthService} from "../../../../services/auth.service";
import OBJECT_TYPES from "../../../../constants/object-types.constants";
import { CommonModule } from '@angular/common';
import {NzFormModule} from "ng-zorro-antd/form";
import {SharedModule} from "../../../../shared-module/shared.module";
import {NzInputModule} from "ng-zorro-antd/input";
import {lastValueFrom} from "rxjs";

@Component({
    selector: 'app-analyses-browse',
    imports: [
        CommonModule,
        NzFormModule,
        SharedModule,
        FormsModule,
        ReactiveFormsModule,
        NzModalModule,
        DynamicBrowseComponent,
        NzInputModule,
    ],
    templateUrl: './analyses-browse.component.html',
    styleUrls: ['./analyses-browse.component.sass']
})
export class AnalysesBrowseComponent implements OnInit, AfterViewInit, OnDestroy, IView {

  BREADCRUMB_NAME = 'METADATA.ANALYSIS.BROWSER.BREADCRUMB';

  @ViewChild(DynamicBrowseComponent)
  private dynamicBrowser: DynamicBrowseComponent;

  pageMeta: any = {
    'pageIndex': 1,
    'pageSize': 10,
    'checkedItems': new Set<number>(),
    'checkedAll': false,
    'selectedFields': [],
  };
  pageData: any[] = [];
  totalSize: number = 0;

  searchValue: string = '';
  loading: boolean = true;

  showForm = false;
  dataForm: UntypedFormGroup = new UntypedFormGroup({
    program: new UntypedFormControl(null, Validators.required),
    programversion: new UntypedFormControl(null, Validators.required),
    name: new UntypedFormControl(),
    algorithm: new UntypedFormControl(),
    description: new UntypedFormControl(),
  });

  constructor(
    private backendService: BackendService,
    private logService: MessageLogService,
    private modal: NzModalService,
    private stateService: StateService,
    private router: Router,
    private activeRoute: ActivatedRoute,
    public readonly authService: AuthService,
    private readonly internacionalizationService: InternationalizationService,
  ) {
  }

  ngOnInit(): void {
    this.search();
  }

  async ngAfterViewInit(): Promise<void> {
    this.loading = true;
    const getState = await this.stateService.getStateCurrenView();
    if (getState) {
      const dynamicTable = this.dynamicBrowser.dynamicTable
      if (getState.pageIndex) {
        dynamicTable.pageIndex = getState.pageIndex;
      }
      if (getState.pageSize) {
        dynamicTable.pageSize = getState.pageSize;
      }
      if (getState.checkedItems && getState.checkedItems instanceof Array) {
        dynamicTable.checkedItems = new Set(getState.checkedItems);
      }
      this.dynamicBrowser.changePage();
    }
    this.loading = false;
  }

  ngOnDestroy(): void {
    const dynamicTable = this.dynamicBrowser.dynamicTable;
    this.stateService.setStateCurrentView({
      pageIndex: dynamicTable.pageIndex,
      pageSize: dynamicTable.pageSize,
      checkedItems: Array.from(dynamicTable.checkedItems),
    });
  }

  changePage() {
    this.search();
  }

  showModal(): void {
    this.showForm = true;
  }

  handleOk(): void {
    this.createAnalisis();
  }

  handleCancel(): void {
    this.showForm = false;
  }

  createAnalisis() {
    this.loading = true;
    this.backendService.postAnalyses(this.dataForm.value)
      .subscribe(
        {next:response => {
          this.logService.addIssues(response['issues']);
        }, error: error => {
          this.loading = false;
          this.logService.addIssues(error['issues']);
        }, complete: () => {
          this.loading = false;
          this.showForm = false;
          this.search();
          this.dynamicBrowser.dynamicTable.checkedItems.clear();
          this.dynamicBrowser.changePage();
        }});
  }

  item2str(item): string {
    return JSON.stringify(item, (k, v) => (k === 'taxa') ? undefined : v);
  }

  searchEvent() {
    this.pageMeta.pageIndex = 1;
    this.pageMeta.checkedItems.clear();
    this.search();
  }

  search() {
    this.loading = true;
    this.backendService.getAnalyses(undefined, {
      'filter': {}, 'order': {}, 'searchValue': this.searchValue,
      'pagination': {'pageIndex': this.pageMeta.pageIndex, 'pageSize': this.pageMeta.pageSize}
    }).subscribe(
      {next: response => {
        this.pageData = response['content'];
        this.totalSize = response['count'];
        this.logService.addIssues(response['issues']);
      }, error: error => {
        this.loading = false;
        this.logService.addIssues(error['issues']);
      }, complete: () => {
        this.loading = false;
      }});
  }

  // OPEN
  openItem(item) {
    if (!this.authService.havePermission('gui-analysis-read')) { return; }
    let url = '';
    if (item.object_type) {
      url = OBJECT_TYPES[item.object_type].pageDetail;
    }
    this.router.navigate([url ?? 'analysisDetail', item.analysis_id]);
  }

  // DELETE
  async deleteSelection(event) {
    if (!event.filter.analysis_id) {
      return;
    }
    this.modal.confirm({
      nzTitle: `<i>${await this.internacionalizationService.translate('METADATA.ANALYSIS.BROWSER.DELETE_MODAL_SELECTION.TITLE')}</i>`,
      nzContent: `<b>${await this.internacionalizationService.translate('METADATA.ANALYSIS.BROWSER.DELETE_MODAL_SELECTION.CONTENT')} </b>${event.filter.analysis_id.unary.length}`,
      nzCentered: true,
      nzOnOk: async () => {
        try {
          this.loading = true;
          const response: any = await lastValueFrom(this.backendService.deleteAnalyses(undefined, event), { defaultValue: undefined });
          this.logService.addIssues(response.issues);
          this.dynamicBrowser.dynamicTable.checkedItems.clear();
        } catch (e) {
          if (e.issues) {
            this.logService.addIssues(e.issues);
          }
        }
        this.search();
        this.dynamicBrowser.changePage();
        this.loading = false;
      }
    });
  }

  async deleteItem(event) {
    this.modal.confirm({
      nzTitle: `<i>${await this.internacionalizationService.translate('METADATA.ANALYSIS.BROWSER.DELETE_MODAL_ITEM.TITLE')}</i>`,
      nzContent: this.item2str(event),
      nzCentered: true,
      nzOnOk: async () => {
        this.loading = true;
        try {
          const response: any = await lastValueFrom(this.backendService.deleteAnalyses(event.analysis_id), { defaultValue: undefined });
          this.logService.addIssues(response.issues);
        } catch (e) {
          if (e.issues) {
            this.logService.addIssues(e.issues);
          }
        }
        this.search();
        this.dynamicBrowser.changePage();
        this.loading = false;
      }
    });
  }

}
