import { Component, OnDestroy, OnInit, ViewEncapsulation } from '@angular/core';
import {
  AbstractControl,
  AbstractControlOptions,
  FormsModule, ReactiveFormsModule,
  UntypedFormBuilder,
  UntypedFormGroup,
  ValidationErrors
} from '@angular/forms';
import { ActivatedRoute } from '@angular/router';
import { AuthService } from 'src/app/services/auth.service';
import { BackendService } from 'src/app/services/backend.service';
import { StateService } from 'src/app/services/state.service';
import proj4 from 'proj4';
import IView from '@interfaces/view.interfaces';
import { CommonModule } from '@angular/common';
import {InfoModalComponent} from "../../../../components/miscellaneous/info-modal/info-modal.component";
import {SharedModule} from "../../../../shared-module/shared.module";
import {DisableControlDirective} from "../../../../directives/disable-control.directive";
import {NzFormModule} from "ng-zorro-antd/form";
import {NzSpinModule} from "ng-zorro-antd/spin";
import {BackButtonComponent} from "../../../../components/miscellaneous/back-button/back-button.component";
import {NzPageHeaderModule} from "ng-zorro-antd/page-header";
import {lastValueFrom} from "rxjs";

@Component({
    selector: 'app-crs-detail',
    imports: [
        CommonModule,
        InfoModalComponent,
        SharedModule,
        DisableControlDirective,
        NzFormModule,
        FormsModule,
        ReactiveFormsModule,
        NzSpinModule,
        BackButtonComponent,
        NzPageHeaderModule,
    ],
    templateUrl: './crs-detail.component.html',
    styleUrls: ['./crs-detail.component.sass'],
    encapsulation: ViewEncapsulation.None
})
export class CrsDetailComponent implements OnInit, OnDestroy, IView {

  BREADCRUMB_NAME = {
    key: 'METADATA.CRS.DETAIL.BREADCRUMB',
    params: {
      name: '',
    }
  };

  id;
  name;
  formGroup: UntypedFormGroup;
  loading = false;
  formGroupChange = false;
  isVisibleModalUpdateError = false;

  constructor(
    private readonly fb: UntypedFormBuilder,
    public readonly authService: AuthService,
    private readonly activatedRouter: ActivatedRoute,
    private readonly stateService: StateService,
    private readonly backendService: BackendService,
  ) { }

  async ngOnInit(): Promise<void> {
    this.loading = true;
    const crsSnapshot = this.activatedRouter.snapshot.data.crs;
    this.id = crsSnapshot.id;
    this.name = crsSnapshot.name;
    this.initFormGroup();
    const getState = await this.stateService.getStateCurrenView();
    if (getState && getState.formGroupChange) {
      this.setFormValueByStateService(getState);
    } else {
      this.setFormValueByResponse(crsSnapshot);
    }
    this.loading = false;
  }

  ngOnDestroy(): void {
    this.BREADCRUMB_NAME.params.name = this.name;
    if (!this.formGroupChange) {
      this.stateService.setStateCurrentView({
        formGroupChange: this.formGroupChange,
      }).then();
    } else {
      const value = {
        name: this.formGroup.get('name').value,
        code: this.formGroup.get('code').value,
        proj4: this.formGroup.get('proj4').value,
        formGroupChange: this.formGroupChange,
      };
      this.stateService.setStateCurrentView(value).then();
    }
  }

  initFormGroup(): void {
    this.formGroup = this.fb.group({
      id: [{ value: this.id, disabled: true }],
      name: [''],
      code: [''],
      proj4: [''],
    }, {
      validators: [
        this.nameValidator.bind(this),
        this.codeValidator.bind(this),
        this.proj4Validator.bind(this)
      ],
    } as AbstractControlOptions);
    this.formGroup.valueChanges.subscribe({next:(change) => this.formGroupChange = true});
  }

  setFormValueByResponse(response) {
    this.name = response.name;
    this.formGroup.get('name').setValue(response.name ? response.name : '');
    this.formGroup.get('code').setValue(response.attributes && response.attributes.code ? response.attributes.code : '');
    this.formGroup.get('proj4').setValue(response.attributes && response.attributes.proj4 ? response.attributes.proj4 : '');
  }

  async setFormValueByStateService(state): Promise<void> {
    if (state.name) {
      this.formGroup.get('name').setValue(state.name);
    }
    if (state.code) {
      this.formGroup.get('code').setValue(state.code);
    }
    if (state.proj4) {
      this.formGroup.get('proj4').setValue(state.proj4);
    }
    if (state.formGroupChange) {
      this.formGroupChange = state.formGroupChange;
    }
  }

  async modifyItem(): Promise<void> {
    this.loading = true;
    try {
      const value = {
        name: this.formGroup.get('name').value,
        attributes: {
          code: this.formGroup.get('code').value,
          proj4: this.formGroup.get('proj4').value
        }
      }
      const response: any = await lastValueFrom(this.backendService.putHierarchyNodes(this.id, value), { defaultValue: undefined });
      this.name = value.name;
    } catch (e) {
      this.isVisibleModalUpdateError = true;
    }
    this.loading = false;
  }

  async cancelModifyItem(): Promise<void> {
    this.loading = true;
    try {
      const response: any = await lastValueFrom(this.backendService.getHierarchyNodes(this.id), { defaultValue: undefined });
      this.setFormValueByResponse(response.content);
    } catch (e) {

    }
    this.loading = false;
    this.formGroupChange = false;
  }

  isDisableEdit(): boolean {
    return !this.authService.havePermission('gui-crs-edit');
  }

  // Validators
  private nameValidator(control: AbstractControl): ValidationErrors | null {
    let error = false;
    let emptyError = false;
    const name = control.get('name').value;
    if (!name || name.trim() === '') {
      error = true;
      emptyError = true;
    }
    return error ? {
      nameError: {
        empty: emptyError,
      }
    } : null;
  }

  getMessageNameError(): string[] | null {
    const message = [];
    if (!this.formGroup.errors || !this.formGroup.errors.nameError) {
      return null;
    }
    if (this.formGroup.errors.nameError.empty) {
      message.push('METADATA.CRS.DETAIL.NAME_INPUT.ERROR_EMPTY');
    }
    return message;
  }

  private codeValidator(control: AbstractControl): ValidationErrors | null {
    let error = false;
    let emptyError = false;
    const code = control.get('code').value;
    if (!code || code.trim() === '') {
      error = true;
      emptyError = true;
    }
    return error ? {
      codeError: {
        empty: emptyError,
      }
    } : null;
  }

  getMessageCodeError(): string[] | null {
    const message = [];
    if (!this.formGroup.errors || !this.formGroup.errors.codeError) {
      return null;
    }
    if (this.formGroup.errors.codeError.empty) {
      message.push('METADATA.CRS.DETAIL.CODE_INPUT.ERROR_EMPTY');
    }
    return message;
  }

  proj4ValidatorFormat(proj4Value): boolean {
    try {
      proj4.defs('ESPG:1232', proj4Value);
      return true;
    } catch (e) {
      return false;
    }
  }

  private proj4Validator(control: AbstractControl): ValidationErrors | null {
    let error = false;
    let errorCode: any = {
      proj4Error: {
      }
    }
    const proj4Value = control.get('proj4').value;
    if (!proj4Value || proj4Value.trim() === '') {
      error = true;
      errorCode.proj4Error.empty = true;
    } else if (!this.proj4ValidatorFormat(proj4Value)) {
      error = true;
      errorCode.proj4Error.format = true;
    }
    return error ? errorCode : null;
  }

  getMessageProj4Error(): string[] | null {
    const message = [];
    if (!this.formGroup.errors || !this.formGroup.errors.proj4Error) {
      return null;
    }
    if (this.formGroup.errors.proj4Error.empty) {
      message.push('METADATA.CRS.DETAIL.PROJ4_INPUT.ERROR_EMPTY');
    }
    if (this.formGroup.errors.proj4Error.format) {
      message.push('METADATA.CRS.CREATE_MODAL.PROJ4_INPUT.ERROR_FORMAT');
    }
    return message;
  }

}
