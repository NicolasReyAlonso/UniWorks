  constructor(
    private readonly filesService: FilesService,
    private readonly socketService: SocketService,

  ) {
    super(filesService);

    this.BREADCRUMB_NAME.params.name = `${job.id} - ${job.process.name}`;
  }

  async ngOnInit(): Promise<void> {
    this.job = this.activatedRoute.snapshot.data.job;
    this.process = this.job.process;
    const resources: any[] = this.activatedRoute.snapshot.data.resources["content"];
    this.resource = resources.find(value => {
      return this.job.resource_id === value.id;
    }).name;
    this.options = this.process.schema_inputs.options;
    this.subprocesses = [];
    this.form = new UntypedFormGroup({});
    this.changeProgressStatus(this.job.status);
    this.model = JSON.parse(JSON.parse(this.job.inputs)).parameters.script_params;
    if (this.options) {
      if (!('formState' in this.options))
        this.options['formState'] = {}
      this.options['formState']['mainModel'] = this.model;
    }

    let resultFiles;
    const processes = this.DEBUG_JSON_FORMLY_FIELDS ?? this.process.schema_inputs.fields;
    // Gives a field an onChange function
    const initField = function(field){
      if (field.props && field.props.changeExpr !== undefined) {
        field.props.changeFunc = Function('$event', 'field', field.props.changeExpr).bind(this);
      }
      if (field.props.initExpr !== undefined) {
        this.initializeFields(field.props.initExpr, field, this.getModelValue(field.key, this.model));
      }
      return field;
    }.bind(this);
    for (const [i, subprocess] of processes.entries()) {
      if ('advanced_parameters' === subprocess.fieldGroup.at(-1).key) {
          let newFieldGroup;
          newFieldGroup = subprocess.fieldGroup.slice(0, -1).map(initField);
          subprocess.fieldGroup.at(-1).fieldGroup = subprocess.fieldGroup.at(-1).fieldGroup.map(initField)
          newFieldGroup.push(subprocess.fieldGroup.at(-1));
          processes[i].fieldGroup = newFieldGroup;
        } else {
          processes[i].fieldGroup = subprocess.fieldGroup.map(initField);
        }
    }
    this.fields = processes;
    let importedObjects = {};
    let object_type: string;
    const import_object_types = {
      'phylotrees': [],
      'layers': [],
      'alignments': []
    };
    for (const subprocess of this.fields) {
      for (const field of subprocess.fieldGroup) {
        if (field.key === 'advanced_parameters') {
          for (const adv_p of field.fieldGroup) {
            adv_p.props.disabled = true;
          }
        }
        else
          field.props.disabled = true;
      }

      resultFiles = [];
      for (const output of JSON.parse(this.job.outputs)) {
        const outputMap = JSON.parse(output);
        if (outputMap.subprocess === subprocess.key) {
          object_type = Object.values(outputMap['object_type'])[0].toString();
          if (object_type in import_object_types) {
            if (!(object_type in importedObjects)) {
              importedObjects[object_type] = await this.getImportedObjects(object_type,
                {'job_id': this.job.id.toString()})
            }

            outputMap['imported'] = false;
            outputMap['analysis_id'] = -1;

            for (const obj of importedObjects
              [object_type]) {
              if (obj['sourceuri'].split("/").at(-1) == outputMap['file']) {
                outputMap['imported'] = true;
                outputMap['analysis_id'] = obj['analysis_id']
              }
            }
          }
          resultFiles.push(outputMap);
        }
      }

      this.subprocesses.push({
        schemaInputs: [subprocess],
        formlyModel: this.model,
        name: subprocess.key,
        outputs: resultFiles
      });
    }
    this.processStatusChangeSubscription = this.socketService.processStatusChangeSubject.subscribe({next:(event) => {
      if (this.job.id === event.id) {
        this.changeProgressStatus(event.status);
      }
    }});

    //IMPORT MODAL

    this.importModalFields = [
      {
        key: 'name',
        type: 'input',
        props: {
          required: true,
          label: await this.internationalizationService.translate("PROCESSES.PROCESSES_DETAIL.IMPORT_MODAL.NAME"),
        }
      }
    ];
  }

  private async getImportedObjects(object_type: string,filter_dict: {}): Promise<{}> {
    const filter = this.backendService.filter(filter_dict)
    let response: ArrayBuffer;
    if (object_type == 'alignments') {
      response = await lastValueFrom(this.backendService.getAlignments(null, filter), { defaultValue: undefined });
    } else if (object_type == 'phylotrees') {
      response = await lastValueFrom(this.backendService.getPhylotrees(null, filter), { defaultValue: undefined });
    } /*else if (object_type == 'layers') {
      response = await lastValueFrom(this.backendService.getLayerGIS(null, filter), { defaultValue: undefined });
    } */else {
      return [];
    }

    return response['content'];
  }

  private changeProgressStatus(status: string) {
    this.step_description = status;
    if (status === 'error' || status === 'cancelling' || status === 'cancelled') {
      this.progressPercentage = 100;
      this.progressStatus = 'exception';
      this.sidebarService.sidebarEventEmitter.emit({
        type: "customRawUrlSelect",
        value: "/process/processesBrowseDone?status=done"
      });
    } else if (status === 'success') {
      this.progressPercentage = 100;
      this.progressStatus = 'success';
      this.sidebarService.sidebarEventEmitter.emit({
        type: "customRawUrlSelect",
        value: "/process/processesBrowseDone?status=done"
      });
    } else if (status === 'manage_error') {
      this.progressStatus = 'exception';
      this.sidebarService.sidebarEventEmitter.emit({
        type: "customRawUrlSelect",
        value: "/process/processesBrowseToDo?status=todo"
      });
    } else {
      this.progressPercentage = ((ProcessDetailComponent.STEPS.indexOf(status)) /
        (ProcessDetailComponent.STEPS.length - 1)) * 100;
      this.progressStatus = 'normal';
      this.sidebarService.sidebarEventEmitter.emit({
        type: "customRawUrlSelect",
        value: "/process/processesBrowseToDo?status=todo"
      });
    }
  }

  ngOnDestroy() {
    this.processStatusChangeSubscription.unsubscribe();
  }

  progressFormat(percent: number) {

    if (percent === 100) {
      return 'Finished';
    } else {
      const step = (percent / 100) * (ProcessDetailComponent.STEPS.length - 1);
      return Math.round(step).toString() + ' / ' + (ProcessDetailComponent.STEPS.length - 1);
    }
  }

  downloadFile(fileName: string) {
    this.filesService.downloadJobResult(this.job.id, fileName);
  }

  viewFile(file: any) {
    if (file['object_type']['bos'] == 'alignments' && file['type'] == 'fasta') {
      this.router.navigate(['nextgendemMsaBrowser'], {
        queryParams: {
          url: this.filesService.getFileFromJobRequestUrl(this.job.id, file['file']),
          analysis_id: file['analysis_id'],
        }
      });
    }
    if (file['object_type']['bos'] == 'phylotrees' && file['type'] == 'nexus') {
      this.router.navigate(['archaeopteryxBrowser'], {
        queryParams: {
          url: this.filesService.getFileFromJobRequestUrl(this.job.id, file['file'])
        }
      });
    } else if (file['object_type']['bos'] == 'phylotrees' && file['type'] == 'newick') {
      this.router.navigate(['archaeopteryxBrowser'], {
        queryParams: {}
      });
    }

    if (file['object_type']['bos'] == 'blast' && file['type'] == 'blast') {
      this.router.navigate(['blasterPage'], {
        queryParams: {
          requestUrl: this.filesService.getFileFromJobRequestUrl(this.job.id, file['file']),
        }
      });
      return;
    }

    if (file['object_type']['bos'] == 'mold' && file['type'] == 'mold') {
      let alnUrl = '';
      for (let subprocess of this.subprocesses) {
        if ('MOLD' in subprocess.formlyModel) {
          let aln = subprocess.formlyModel['MOLD']['alignment_selector'][0]['object_type'];
          if ('filesAPI' in aln) {
            alnUrl = this.filesService.getFileFromPathUrl(aln['filesAPI']);
          }
        }

      }
      this.router.navigate(['ngdBarconding'], {
        queryParams: {
          requestUrl: this.filesService.getFileFromJobRequestUrl(this.job.id, file['file']),
          alnUrl: alnUrl
        }
      });
      return;
    }
  }

  canViewFile(file: any) {
    let canView = false;
    if ('bos' in file['object_type']) {
      if (file['object_type']['bos'] == 'alignments' && file['type'] == 'fasta') {
        canView = true;
      } else if (file['object_type']['bos'] == 'phylotrees' && file['type'] == 'nexus') {
        canView = true;
      } else if (file['object_type']['bos'] == 'phylotrees' && file['type'] == 'newick') {
         canView = true;
      } else if (file['object_type']['bos'] == 'blast' && file['type'] == 'blast') {
        canView = true;
      } else if (file['object_type']['bos'] == 'mold' && file['type'] == 'mold') {
        canView = true;
      }
    }
    return canView;
  }

  import(resultFile: {}, importModalOutput: {}) {
    this.isImporting = resultFile['file'];
    resultFile['bos_name'] = importModalOutput['name'] ?? '';
    this.filesService.importJobResult(this.job.id, resultFile).subscribe({
      next:response => {
        this.isImporting = undefined;
        this.msg.info(`Importando ${resultFile['file']} como ${resultFile['object_type'].toString()}`);
      }, error: error => {
        this.isImporting = undefined;
        this.msg.error(error.error.message + `\nError al importar ${resultFile['file']} como ${resultFile['object_type'].toString()}`);
        this.notificationService.createNotificationWithType(TypeNotificationEnum.error, 'Error',
          `${resultFile['file']} not imported correctly`, 'bottomRight');
      }, complete: async () => {
        resultFile['imported'] = true;
        const content = await this.getImportedObjects(Object.values(resultFile['object_type'])[0].toString(),
          {sourceuri: '/jobs/' + this.job.id + '/' + resultFile['file']});
        resultFile['analysis_id'] = content[0]['analysis_id'];
        this.isImporting = undefined;
        this.msg.success(`${resultFile['file']} importado correctamente como ${resultFile['object_type'].toString()}`);
        await this.notificationService.createNotificationWithType(TypeNotificationEnum.success, 'Success',
          `${resultFile['file']} imported correctly`, 'bottomRight');
      }});
  }

  async onClickCancelProcess(): Promise<void> {
    this.contentParamsCancelProcessModal.name = this.job.id;
    this.isVisibleCancelProcessModal = true;
  }

  async cancelProcess(): Promise<void> {
    this.isVisibleCancelProcessModal = false;
    try {
      const response: any = await lastValueFrom(this.backendService.putJob(this.job.id, { "status": "cancelling" }), { defaultValue: undefined });
      this.changeProgressStatus("cancelling");
      if (response.issues) {
        this.msg.addIssues(response.issues);
      }
      await this.notificationService.createNotificationWithType(
        TypeNotificationEnum.success,
        "PROCESSES.PROCESSES_DETAIL.CANCEL_PROCESS_NOTIFICATION_CORRECTLY.TITLE",
        "PROCESSES.PROCESSES_DETAIL.CANCEL_PROCESS_NOTIFICATION_CORRECTLY.CONTENT",
        "bottomRight"
      );
    } catch (err) {
      if (err.issues) {
        this.msg.addIssues(err.issues);
      }
      await this.notificationService.createNotificationWithType(
        TypeNotificationEnum.error,
        "PROCESSES.PROCESSES_DETAIL.CANCEL_PROCESS_NOTIFICATION_ERROR.TITLE",
        "PROCESSES.PROCESSES_DETAIL.CANCEL_PROCESS_NOTIFICATION_ERROR.CONTENT",
        "bottomRight"
      );
    }
  }

  showCancelButton(): boolean {
    return this.progressStatus !== 'success' &&
      this.progressStatus !== 'exception'  &&
      this.authService.havePermission('gui-job-cancel');
  }

  async goToDetail(resultFile: {}) {
    await this.router.navigate(['alignmentDetail', resultFile['analysis_id']]);
  }

  DEBUG_JSON_FORMLY_FIELDS: any = null;

}
