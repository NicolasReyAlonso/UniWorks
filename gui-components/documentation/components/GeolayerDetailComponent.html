  constructor(
    private readonly activeRoute: ActivatedRoute,
    private readonly fb: UntypedFormBuilder,
    private readonly nzModalService: NzModalService,
    private readonly backendService: BackendService,
    private readonly messageLogService: MessageLogService,
    private readonly notificationService: NotificationService,
    private readonly router: Router,
    public readonly authService: AuthService,
    private readonly stateService: StateService,
  ) {
    this.layer = this.activeRoute.snapshot.data.layer.content;
    this.BREADCRUMB_NAME.params.layer = this.layer.name;
  }

  async ngOnInit(): Promise<void> {
    this.loading = true;
    this.initForm();
    this.formGroup.valueChanges.subscribe({next:(change) => this.formValueChange(change)});
    const activatedRouterDataStyles = this.activeRoute.snapshot.data.styles;
    this.styles = [];
    if (activatedRouterDataStyles) {
      Object.keys(activatedRouterDataStyles).forEach((key) => {
        this.styles.push({ style: key, value: activatedRouterDataStyles[key] });
      });
    }
    await this.startFormValues();
    this.loading = false;
  }

  async ngOnDestroy(): Promise<void> {
    if (!this.formGroupChange) {
      this.stateService.setStateCurrentView({
        formGroupChange: this.formGroupChange,
      }).then();
    } else {
      const value = {
        name: this.formGroup.get('name').value,
        tags: this.formGroup.get('tags').value,
        authors: this.formGroup.get('authors').value,
        themes: this.formGroup.get('themes').value,
        crs: this.formGroup.get('crs').value,
        creation_date: this.formGroup.get('creation_date').value,
        source: this.formGroup.get('source').value,
        case_studies: this.formGroup.get('case_studies').value,
        formGroupChange: this.formGroupChange,
      };
      this.stateService.setStateCurrentView(value).then();
    }
  }

  initForm() {
    this.caseStudies = this.activeRoute.snapshot.data.caseStudies;
    this.themes = this.activeRoute.snapshot.data.themes;
    this.crs = this.activeRoute.snapshot.data.crs;
    this.sources = this.activeRoute.snapshot.data.sources;
    this.layer = this.activeRoute.snapshot.data.layer.content;
    this.formGroup = this.fb.group({
      id: this.fb.control({value: this.layer.id, disabled: true}),
      name: this.fb.control(''),
      tags: this.fb.control([]),
      authors: this.fb.control(''),
      themes: this.fb.control([]),
      crs: this.fb.control(null),
      creation_date: this.fb.control(this.layer.attributes.creation_date ? this.layer.attributes.creation_date : null),
      types: this.fb.control([]),
      case_studies: this.fb.control([]),
      source: this.fb.control([]),
      file_name: this.fb.control({
        value: this.layer.attributes.file_name ? this.layer.attributes.file_name : null,
        disabled: true
      }),
    });
    this.formGroupAdvance = this.fb.group({
      wks: [{value: this.layer.wks, disabled: true}],
      wms_url: [{value: this.layer.wms_url, disabled: true}],
      geoserver_name: [{value: this.layer.geoserver_name, disabled: true}],
      layer_type: [{value: this.layer.layer_type, disabled: true}],
      properties: [this.layer.properties ? this.layer.properties : []],
    });
    if (!this.layer.attributes.instance_id) {
      this.formGroup.addControl('source', this.fb.control(null));
    } else {
      this.formGroup.addControl('instance_id', this.fb.control({
        value: this.layer.attributes.instance_id,
        disabled: true
      }));
    }
  }

  async startFormValues(): Promise<void> {
    const getState = await this.stateService.getStateCurrenView();
    if (!getState || !getState.formGroupChange) {
      await this.loadData();
      this.formGroupChange = false;
    } else {
      console.log(getState);
      if (getState.name) {
        this.formGroup.get('name').setValue(getState.name);
      }
      if (getState.tags) {
        this.formGroup.get('tags').setValue(getState.tags);
      }
      if (getState.authors) {
        this.formGroup.get('authors').setValue(getState.authors);
      }
      if (getState.themes) {
        this.formGroup.get('themes').setValue(getState.themes);
      }
      if (getState.crs) {
        this.formGroup.get('crs').setValue(getState.crs);
      }
      if (getState.creation_date) {
        this.formGroup.get('creation_date').setValue(getState.creation_date);
      }
      if (getState.source) {
        this.formGroup.get('source').setValue(getState.source);
      }
      if (getState.case_studies) {
        this.formGroup.get('case_studies').setValue(getState.case_studies);
      }
      if (getState.formGroupChange) {
        this.formGroupChange = getState.formGroupChange;
      }
    }
  }

  ngAfterViewInit(): void {
    this.recreateGeoViewer();
    this.annotationsForm.object_uuid = this.layer.uuid;
    this.annotationsForm.loadAnnotations();
    const addAttributeModal = this.nextgendemGeolayerComponent.nextgendemGeolayerAddAttributeModalComponent
    addAttributeModal.uuidLayerFilter = [ this.layer.uuid ]
  }

  recreateGeoViewer() {

  }

  openExportLayerModal() {
    this.geolayerExportModal.openModal([this.layer]);
  }

  private formValueChange(change) {
    this.formGroupChange = true;
  }

  private async loadData(layer?) {
    this.loading = true;
    this.formGroupChange = false;
    this.layer = layer ? layer : this.layer;
    let firstPropertiesNumeric = null;
    for (const property of this.layer.properties) {
      if (property.type === 'numeric') {
        firstPropertiesNumeric = property;
        break;
      }
    }
    const caseStudies2Field = [];
    for (const caseStudy of this.layer.case_studies) {
      caseStudies2Field.push(caseStudy.case_study_id);
    }
    this.layer.case_studies = caseStudies2Field;
    if (this.layer.name) {
      this.formGroup.get('name').setValue(this.layer.name);
    }
    if (this.layer.attributes.tags) {
      this.formGroup.get('tags').setValue(this.layer.attributes.tags);
    }
    if (this.layer.attributes.authors) {
      this.formGroup.get('authors').setValue(this.layer.attributes.authors);
    }
    if (this.layer.attributes.themes_id) {
      this.formGroup.get('themes').setValue(this.layer.attributes.themes_id);
    }
    if (this.layer.attributes.crs_id) {
      this.formGroup.get('crs').setValue(this.layer.attributes.crs_id);
    }
    if (this.layer.attributes.creation_date) {
      this.formGroup.get('creation_date').setValue(this.layer.attributes.creation_date);
    }
    if (this.layer.types) {
      this.formGroup.get('types').setValue(this.layer.types);
    }
    if (this.layer.case_studies) {
      this.formGroup.get('case_studies').setValue(this.layer.case_studies);
    }
    if (this.layer.attributes.file_name) {
      this.formGroup.get('file_name').setValue(this.layer.attributes.file_name);
    }
    if (!this.layer.attributes.instance_id) {
      if (this.layer.attributes.source_id) {
        this.formGroup.get('source').setValue(this.layer.attributes.source_id);
      }
    }
  }

  getCaseStudyById(id): any {
    return this.caseStudies.find((value) => value.id == id);
  }

  addTags(inputName: HTMLInputElement): void {
    const name = inputName.value;
    if (!name || name === '') {
      return;
    }
    const tag = this.fb.group({
      name: [name, Validators.required],
    });
    const controlTags = this.formGroup.get('tags');
    controlTags.setValue([name, ...controlTags.value]);
    inputName.value = '';
  }

  removeTag(event, i): void {
    event.preventDefault();
    event.stopPropagation();
    const controlTags = this.formGroup.get('tags');
    const newArray = [...controlTags.value];
    newArray.splice(i, 1);
    controlTags.setValue(newArray);
  }

  openCaseStudy(id): void {
    if (this.authService.havePermission('gui-case-study-read')) {
      this.router.navigate(['caseStudiesDetail', id]);
    }
  }

  openAddModalCaseStudies(): void {
    this.selectModal = [];
    this.addCaseStudiesVisibleModal = true;
  }

  getAddCaseStudiesOptions(): any {
    const caseStudiesValue = this.formGroup.get('case_studies').value as any[];
    return this.caseStudies.filter((value) => !caseStudiesValue.includes(value.id));
  }

  addCaseStudiesModal() {
    const controlCaseStudies = this.formGroup.get('case_studies');
    controlCaseStudies.setValue([...controlCaseStudies.value, ...this.selectModal]);
    this.selectModal = [];
    this.addCaseStudiesVisibleModal = false;
  }

  removeCaseStudies(index) {
    const controlCaseStudies = this.formGroup.get('case_studies');
    const newArray = [...controlCaseStudies.value];
    newArray.splice(index, 1);
    controlCaseStudies.setValue(newArray);
  }

  async modifyLayer(): Promise<void> {
    this.loading = true;
    const value = {
      name: this.formGroup.get('name').value,
      types: this.formGroup.get('types').value,
      case_studies: this.formGroup.get('case_studies').value,
      attributes: {
        tags: this.formGroup.get('tags').value,
        authors: this.formGroup.get('authors').value,
        themes_id: this.formGroup.get('themes').value,
        crs_id: this.formGroup.get('crs').value,
        creation_date: this.formGroup.get('creation_date').value,
        file_name: this.formGroup.get('file_name').value,
        source_id: this.formGroup.get('source').value,
      },
    };
    if (!this.layer.attributes.instance_id) {
      value.attributes.source_id = this.formGroup.get('source').value;
    }
    console.log(value);
    try {
      const putLayerResponse: any = await lastValueFrom(this.backendService.putLayerGIS(this.layer.id, value), { defaultValue: undefined });
      this.messageLogService.addIssues(putLayerResponse.issues);
      this.notificationService.createNotificationWithType(TypeNotificationEnum.success, 'Capa actualizada', 'Se ha actualizado la capa correctamente', 'bottomRight');
      this.loading = false;
      this.formGroupChange = false;
    } catch (err) {
      this.nzModalService.error({
        nzTitle: 'Error',
        nzContent: 'No se ha conseguido actualizar la capa.',
        nzCentered: true,
      });
      this.loading = false;
    }
  }

  async cancelModifyLayer(): Promise<void> {
    try {
      const layerResponse: any = await lastValueFrom(this.backendService.getLayerGIS(this.layer.id), { defaultValue: undefined });
      const layerResponseContent = layerResponse.content;
      await this.loadData(layerResponseContent);
      this.formGroupChange = false;
      this.loading = false;
    } catch (error) {

    }
  }

  isDisableEdit(): boolean {
    return !this.authService.havePermission('gui-layer-edit');
  }

  onClickEditProperty(data, index): void {
    if (!this.authService.havePermission('gui-layer-edit')) {
      return;
    } else {
      this.propertySelectEditModal = data.name;
      this.propertySelectIndexEditModal = index;
      this.geopropertyEditModal.changeValuesByProperties(data);
      this.geopropertyEditModal.name = this.propertySelectEditModal;
      this.geopropertyEditModal.loading = false;
      this.isVisiblePropertyEditModal = true;
    }
  }

  async acceptGeopropertyEditModal(event): Promise<void> {
    this.geopropertyEditModal.loading = true;
    try {
      const response: any = await lastValueFrom(this.backendService.putLayerProperty(this.layer.id, this.propertySelectEditModal, event), { defaultValue: undefined });
      this.changeValueProperty(event, this.propertySelectIndexEditModal);
      if (response.issues) {
        this.messageLogService.addIssues(response.issues);
      }
      this.isVisiblePropertyEditModal = false;
    } catch (e) {
      this.geopropertyEditModal.isVisibleModalError = true;
      if (e.issues) {
        this.messageLogService.addIssues(e.issues);
      }
    }
    this.recreateGeoViewer();
    this.geopropertyEditModal.loading = false;
  }

  changeValueProperty(value, index: number): void {
    const control = this.formGroupAdvance.get('properties');
    const properties = [...control.value];
    properties[index] = {
      ...properties[index],
      ...value
    };
    control.setValue(properties)
  }

}
