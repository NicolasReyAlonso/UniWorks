import { Location } from '@angular/common';
import {Component, OnDestroy, OnInit, ViewChild} from '@angular/core';
import { ActivatedRoute } from '@angular/router';
import {StateService} from "../../../../services/state.service";
import {IdentityFormComponent} from "src/app/components/admin/identities/identity-form/identity-form.component";
import IView from "../../../../interfaces/view.interfaces";
import { CommonModule } from '@angular/common';
import {BackButtonComponent} from "../../../../components/miscellaneous/back-button/back-button.component";
import {NzPageHeaderModule} from "ng-zorro-antd/page-header";
import {SharedModule} from "../../../../shared-module/shared.module";

@Component({
    selector: 'app-identity-modify',
    imports: [
        CommonModule,
        IdentityFormComponent,
        BackButtonComponent,
        NzPageHeaderModule,
        SharedModule,
    ],
    templateUrl: './identity-modify.component.html',
    styleUrls: ['./identity-modify.component.sass']
})
export class IdentityModifyComponent implements OnInit, OnDestroy, IView {

  BREADCRUMB_NAME = {
    key: 'SECURITY_ADMIN.IDENTITIES.MODIFY.BREADCRUMB',
    params: {
      name: '',
    }
  };

  identityId = null;
  identityData;
  needUpdateState = true;

  stateForm;

  @ViewChild('identityFormElementRef', { static: false}) identityFormElementRef: IdentityFormComponent;

  constructor(
    private readonly activeRoute: ActivatedRoute,
    private readonly location: Location,
    private readonly stateService: StateService,
  ) { }

  async ngOnInit(): Promise<void> {
    this.identityData = this.activeRoute.snapshot.data.identity["content"];
    this.identityId = this.identityData.id;
    const getState = await this.stateService.getStateCurrenView();
    if (getState) {
      this.stateForm = getState;
    }
  }

  ngOnDestroy() {
    this.BREADCRUMB_NAME.params.name = this.identityData.name;
    if (this.needUpdateState) {
      this.stateService.setStateCurrentView({
        name: this.identityFormElementRef.formGroup.get('name').value,
        email: this.identityFormElementRef.formGroup.get('email').value,
        deactivationTime: this.identityFormElementRef.formGroup.get('deactivationTime').value,
        groups: this.identityFormElementRef.formGroup.get('groups').value,
        organizations: this.identityFormElementRef.formGroup.get('organizations').value,
        roles: this.identityFormElementRef.formGroup.get('roles').value,
        can_login: this.identityFormElementRef.formGroup.get('can_login').value,
      }).then();
    }
  }

  cancelEvent() {
    this.needUpdateState = false;
    this.stateService.setStateCurrentView({}).then();
    this.location.back();
  }

  requestEndForm(event : { methodRequest: 'PUT' | 'POST' | 'GET', responseType: 'error' | 'success', data?: any }) {
    switch (event.methodRequest) {
      case 'PUT':
        if (event.responseType == 'success') {
          this.needUpdateState = false;
          this.stateService.setStateCurrentView({}).then();
          this.location.back();
        }
        break;
    }
  }

}
