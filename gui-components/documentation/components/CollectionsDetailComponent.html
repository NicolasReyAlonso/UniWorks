import {AfterViewInit, Component, OnInit, ViewChild} from '@angular/core';
import {FormsModule, ReactiveFormsModule, UntypedFormBuilder, UntypedFormGroup} from "@angular/forms";
import {AuthService} from "../../../../services/auth.service";
import {
  AnnotationsFormComponent
} from "src/app/components/data/annotations/annotations-form/annotations-form.component";
import {StateService} from "../../../../services/state.service";
import {ModalInfoService} from "../../../../services/modal-info.service";
import {BackendService} from "../../../../services/backend.service";
import {ActivatedRoute, Router} from "@angular/router";
import ObjectTypes from "../../../../constants/object-types.constants";
import {forkJoin, lastValueFrom} from "rxjs";
import {MessageLogService} from "../../../../services/message-log.service";
import {CommonModule} from '@angular/common';
import {SharedModule} from "../../../../shared-module/shared.module";
import {NzTabsModule} from "ng-zorro-antd/tabs";
import {NzListModule} from "ng-zorro-antd/list";
import {NzTableModule} from "ng-zorro-antd/table";
import {DisableControlDirective} from "../../../../directives/disable-control.directive";
import {NzFormModule} from "ng-zorro-antd/form";
import {NzButtonModule} from "ng-zorro-antd/button";
import {NzInputModule} from "ng-zorro-antd/input";
import {NzSpinModule} from "ng-zorro-antd/spin";
import {BackButtonComponent} from "../../../../components/miscellaneous/back-button/back-button.component";
import {NzPageHeaderModule} from "ng-zorro-antd/page-header";
import {NzIconModule} from "ng-zorro-antd/icon";
import {
  DynamicEntitiesTableComponent
} from "../../../../components/dynamics/dynamic-entities-table/dynamic-entities-table.component";

@Component({
    selector: 'app-collections-detail',
    imports: [
        CommonModule,
        SharedModule,
        FormsModule,
        ReactiveFormsModule,
        AnnotationsFormComponent,
        BackButtonComponent,
        NzPageHeaderModule,
        NzTabsModule,
        NzListModule,
        NzTableModule,
        DisableControlDirective,
        NzFormModule,
        NzButtonModule,
        NzInputModule,
        NzSpinModule,
        NzIconModule,
        DynamicEntitiesTableComponent,
    ],
    templateUrl: './collections-detail.component.html',
    styleUrls: ['./collections-detail.component.sass']
})
export class CollectionsDetailComponent implements OnInit, AfterViewInit {

  @ViewChild('annotationsForm', {static: false}) annotationsForm: AnnotationsFormComponent;
  @ViewChild(DynamicEntitiesTableComponent) dynamicEntitiesTableComponent: DynamicEntitiesTableComponent;

  item: any;
  id: string;
  selectedIndexTab = 0;
  loadingGeneral = false;
  formGroup: UntypedFormGroup;
  formGroupChange = false;
  typeObjectsList = new Map<any, any>();
  BREADCRUMB_NAME = {
    key: 'MOLECULAR_DATA.COLLECTIONS.DETAIL.BREADCRUMB',
    params: {
      name: '',
    }
  };
  readonly headerEntities = [
    {
      translation: 'MOLECULAR_DATA.COLLECTIONS.DETAIL.INPUTS.ENTITIES.TH.NAME',
      variableName: 'name'
    },
  ];

  constructor(
    private readonly authService: AuthService,
    private readonly fb: UntypedFormBuilder,
    private readonly stateService: StateService,
    private readonly modalInfo: ModalInfoService,
    private readonly backendService: BackendService,
    private readonly activatedRoute: ActivatedRoute,
    private readonly router: Router,
    private readonly messageLogService: MessageLogService,
  ) {
  }

  async ngOnInit(): Promise<void> {
    this.id = this.activatedRoute.snapshot.params.id;
    this.initFormGroup();
  }

  async ngAfterViewInit() {
    this.loadingGeneral = true;
    try {
      this.item = await this.getCollectionBackend();
      this.BREADCRUMB_NAME.params.name = this.item.name;
    } catch (e) {
      this.modalInfo.showModalInfoDefaultError(e);
    }
    const getState = await this.stateService.getStateCurrenView();
    if (!getState || !getState.formGroupChange) {
      await this.loadData(this.item);
      this.formGroupChange = false;
    } else {
      if (getState.name) {
        this.formGroup.get('name').setValue(getState.name);
      }
      if (getState.formGroupChange) {
        this.formGroupChange = getState.formGroupChange;
      }
    }
    this.loadingGeneral = false;
    this.annotationsForm.object_uuid = this.item.uuid;
    this.annotationsForm.loadAnnotations();
  }

  initFormGroup() {
    this.formGroup = this.fb.group({
      name: '',
    });
    this.formGroup.valueChanges.subscribe({next:(change) => this.formValueChange(change)});

  }

  async getCollectionBackend(): Promise<any> {
    const response: any = await lastValueFrom(this.backendService.getCollections(this.id, {
      'no_attach': true,
      'fast_mode': false
    }), { defaultValue: undefined });
    return response.content;
  }

  async loadData(item?): Promise<void> {
    this.loadingGeneral = true;
    try {
      const content = item ?? await this.getCollectionBackend();
      this.item = content;
      this.BREADCRUMB_NAME.params.name = this.item.name;
      this.formGroup.get('name').setValue(content.name);
      this.formGroupChange = false;
      this.typeObjectsList = new Map();
    } catch (e) {
      console.error(e);
      this.modalInfo.showModalInfoDefaultError(e);
    }
    this.loadingGeneral = false;
  }

  formValueChange(change) {
    this.formGroupChange = true;
  }

  isDisableEdit() {
    return !this.authService.havePermission('gui-collection-edit');
  }

  openItem(item) {
    const objectType = ObjectTypes[item.objectTypeName];
    if (!this.authService.havePermission(objectType.readPermission)) {
      return;
    }
    this.router.navigate([objectType.pageDetail, item.nativeId ?? item.id]);
  }

  entitiesListOfDeleteChange(event) {
    this.formGroupChange = true;
  }

  async cancel(): Promise<void> {
    this.loadingGeneral = true;
    try {
      this.dynamicEntitiesTableComponent.listOfDelete = [];
      await this.loadData();
      this.formGroupChange = false;
    } catch (e) {
      console.error(e);
      this.modalInfo.showModalInfoDefaultError(e);
    }
  }

  async modify(): Promise<void> {
    this.loadingGeneral = true;
    const value = {
      name: this.formGroup.get('name').value,
    };
    try {
      const putCollectionResponse: any = await lastValueFrom(this.backendService.putCollection(this.id, value));
      const observers = [];
      this.dynamicEntitiesTableComponent.listOfDelete.forEach((element) => {
        observers.push(this.backendService.deleteCollectionItemByCollectionAndItem(this.id, element));
      });
      if (Object.keys(observers).length !== 0) await lastValueFrom(forkJoin(observers));
      this.dynamicEntitiesTableComponent.listOfDelete = [];
      this.dynamicEntitiesTableComponent.loadData();
      await this.loadData();
      this.messageLogService.addIssues(putCollectionResponse.issues);
    } catch (e) {
      console.error(e);
      this.modalInfo.showModalInfoDefaultError(e);
    }
    this.loadingGeneral = false;
  }

}
