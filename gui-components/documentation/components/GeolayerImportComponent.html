import {Component, EventEmitter, OnInit} from '@angular/core';
import {
  AbstractControl,
  UntypedFormArray,
  UntypedFormBuilder,
  ValidationErrors,
  Validators,
  FormsModule, ReactiveFormsModule
} from '@angular/forms';
import { ActivatedRoute } from '@angular/router';
import IView from '@interfaces/view.interfaces';
import { BackendService } from 'src/app/services/backend.service';
import {NotificationService, TypeNotificationEnum} from 'src/app/services/notification.service';
import { CommonModule } from '@angular/common';
import {SharedModule} from "../../../shared-module/shared.module";
import {NzFormModule} from "ng-zorro-antd/form";
import {NzInputModule} from "ng-zorro-antd/input";
import {NzButtonModule} from "ng-zorro-antd/button";
import {NzUploadFile, NzUploadModule} from "ng-zorro-antd/upload";
import {NzDatePickerModule} from "ng-zorro-antd/date-picker";
import {NzSelectModule} from "ng-zorro-antd/select";
import {DisableControlDirective} from "../../../directives/disable-control.directive";
import {NzTagModule} from "ng-zorro-antd/tag";
import {NzPageHeaderModule} from "ng-zorro-antd/page-header";
import {NzIconModule} from "ng-zorro-antd/icon";
import {Observable} from "rxjs";

@Component({
    selector: 'app-geolayer-import',
    imports: [
        CommonModule,
        SharedModule,
        FormsModule,
        ReactiveFormsModule,
        NzFormModule,
        NzInputModule,
        NzButtonModule,
        NzUploadModule,
        NzDatePickerModule,
        NzSelectModule,
        NzTagModule,
        NzPageHeaderModule,
        DisableControlDirective,
        NzIconModule,
    ],
    templateUrl: './geolayer-import.component.html',
    styleUrls: ['./geolayer-import.component.sass']
})
export class GeolayerImportComponent implements OnInit, IView {

  BREADCRUMB_NAME = 'GEOGRAPHICAL_DATA.LAYERS.IMPORT.BREADCRUMB';

  dataForm = this.fb.group({
    name: this.fb.control(""),
    case_studies: this.fb.control([]),
    tags: this.fb.array([]),
    authors: this.fb.control(""),
    source: this.fb.control(null),
    themes: this.fb.control([]),
    crs: this.fb.control(null),
    creation_date: this.fb.control(null),
    file: this.fb.control(null),
    fileName: this.fb.control(""),
  }, {
    validators: [
      this.nameValidator,
      this.fileValidator,
      this.tagsValidator,
    ]
  });
  firstSubmitDone = false;
  loading = false;

  fileList = [];

  sources;
  themes;
  crs;
  caseStudies;

  constructor(
    private readonly notificationService: NotificationService,
    private readonly backendService: BackendService,
    private readonly fb: UntypedFormBuilder,
    private readonly activatedRouter: ActivatedRoute,
  ) { }

  get tags() {
    return this.dataForm.controls["tags"] as UntypedFormArray;
  }

  ngOnInit(): void {
    this.sources = this.activatedRouter.snapshot.data.sources;
    this.themes = this.activatedRouter.snapshot.data.themes;
    this.crs = this.activatedRouter.snapshot.data.crs;
    this.caseStudies = this.activatedRouter.snapshot.data.caseStudies;
    this.dataForm.get("file").setValue(this.fileList[0]);
    this.dataForm.get('tags').setValue([]);
  }

  private getAttributes(): { tags: string[], [key: string]: any } {
    const formatTags = [];
    for (const tag of this.tags.controls) {
      formatTags.push(tag.value.name);
    }
    const attributes = {
      tags: formatTags,
      authors: this.dataForm.get("authors").value,
      source_id: this.dataForm.get("source").value,
      themes_id: this.dataForm.get("themes").value,
      crs_id: this.dataForm.get("crs").value,
      creation_date: this.dataForm.get("creation_date").value,
      file_name: this.dataForm.get("fileName").value,
    };
    return attributes;
  }

  importLayer(): void {
    this.loading = true;
    this.firstSubmitDone = true;
    if (this.dataForm.errors) {
      this.notificationService.createNotificationWithType(
        TypeNotificationEnum.error,
        "GEOGRAPHICAL_DATA.LAYERS.IMPORT.NOTIFICATION_SERVICE.FAIL_FORM.TITLE",
        "GEOGRAPHICAL_DATA.LAYERS.IMPORT.NOTIFICATION_SERVICE.FAIL_FORM.CONTENT",
        "bottomRight"
      );
      this.loading = false;
      return;
    }

    this.notificationService.createNotificationWithType(
      TypeNotificationEnum.info,
      'GEOGRAPHICAL_DATA.LAYERS.IMPORT.NOTIFICATION_SERVICE.IMPORTING_LAYER.TITLE',
      'GEOGRAPHICAL_DATA.LAYERS.IMPORT.NOTIFICATION_SERVICE.IMPORTING_LAYER.CONTENT',
      'bottomRight'
    );

    this.backendService.postLayerGIS(this.dataForm.get("file").value, {
      name: this.dataForm.get("name").value,
      wks: "bcs",
      attributes: this.getAttributes(),
      case_studies: this.dataForm.get("case_studies").value,
    }).subscribe({next:res => {
      this.fileList = [];
      this.tags.clear();
      this.dataForm.get("name").setValue("");
      this.dataForm.get("case_studies").setValue([]);
      this.dataForm.get("authors").setValue("");
      this.dataForm.get("source").setValue(null);
      this.dataForm.get("themes").setValue([]);
      this.dataForm.get("crs").setValue(null);
      this.dataForm.get("creation_date").setValue(null);
      this.dataForm.get("fileName").setValue("");
      this.loading = false;
      this.firstSubmitDone = false;
      this.notificationService.createNotificationWithType(
        TypeNotificationEnum.success,
        'GEOGRAPHICAL_DATA.LAYERS.IMPORT.NOTIFICATION_SERVICE.IMPORT_LAYER_SUCCESS.TITLE',
        'GEOGRAPHICAL_DATA.LAYERS.IMPORT.NOTIFICATION_SERVICE.IMPORT_LAYER_SUCCESS.CONTENT',
        'bottomRight'
      );
      console.log(res);
    }, error: error => {
      this.loading = false;
      this.notificationService.createNotificationWithType(
        TypeNotificationEnum.error,
        'GEOGRAPHICAL_DATA.LAYERS.IMPORT.NOTIFICATION_SERVICE.FAIL_IMPORT_LAYER.TITLE',
        'GEOGRAPHICAL_DATA.LAYERS.IMPORT.NOTIFICATION_SERVICE.FAIL_IMPORT_LAYER.CONTENT',
        'bottomRight'
      );
    }});
  }

  removeFile(file: NzUploadFile): boolean  {
    this.fileList = [];
    this.dataForm.get("fileName").setValue('');
    this.dataForm.get("file").setValue(null);
    return true
  }

  // Name
  private nameValidator(control: AbstractControl): ValidationErrors | null {
    let error = false;
    let emptyError = false;
    const name = control.get('name').value;
    if (!name || name == "") {
      error = true;
      emptyError = true;
    }
    return error ? {
      nameError: {
        empty: emptyError,
      }
    } : null;
  }

  showNameError(): boolean {
    if (!this.dataForm.errors || !this.dataForm.errors.nameError || !this.firstSubmitDone) {
      return false;
    }
    return true;
  }

  getMessageNameError(): string {
    let message = "";
    if (!this.dataForm.errors || !this.dataForm.errors.nameError || !this.firstSubmitDone) {
      return null;
    }
    if (this.dataForm.errors.nameError.empty) {
      message += "Introduzca un nombre de capa.";
    }

    return message;
  }

  // Tags

  addTags(inputName: HTMLInputElement): void {
    const name = inputName.value;
    if (!name || name === '') {
      return;
    }
    const tag = this.fb.group({
      name: [name, Validators.required],
    });
    this.tags.push(tag);
    inputName.value = '';
  }

  removeTag(event, i): void {
    event.preventDefault();
    event.stopPropagation();
    this.tags.removeAt(i);
  }

  private tagsValidator(control: AbstractControl): ValidationErrors | null {
    const tags: any[] = control.get('tags').value;
    let error = false;
    let duplicateError = false;
    if (!tags) {
      return null;
    }
    for (let i = 0; i < tags.length; i++) {
      if (duplicateError) break;
      for (let z = 0; z < tags.length; z++) {
        if (z == i) continue;
        if (tags[i].name.toLocaleLowerCase() == tags[z].name.toLocaleLowerCase()) {
          error = true;
          duplicateError = true;
          break;
        }
      }
    }
    return error ? {
      tagsError: {
        duplicateError: duplicateError,
      }
    } : null;
  }

  showTagsError(): boolean {
    if (!this.dataForm.errors || !this.dataForm.errors.tagsError || !this.firstSubmitDone) {
      return false;
    }
    return true;
  }

  getMessageTagsError(): string {
    let message = "";
    if (!this.dataForm.errors || !this.dataForm.errors.tagsError || !this.firstSubmitDone) {
      return null;
    }
    if (this.dataForm.errors.tagsError.duplicateError) {
      message += "Hay más de un tag con el mismo nombre.";
    }
    return message;
  }

  // File

  beforeUpload = (file): boolean => {
    this.fileList = [];
    this.fileList = this.fileList.concat(file);
    this.dataForm.get("fileName").setValue(file.name);
    this.dataForm.get("file").setValue(this.fileList[0]);
    return false;
  }

  private fileValidator(control: AbstractControl): ValidationErrors | null {
    let error = false;
    let emptyError = false;
    const file = control.get('file').value;
    if (!file) {
      error = true;
      emptyError = true;
    }
    return error ? {
      fileError: {
        empty: emptyError,
      }
    } : null;
  }

  showFileError(): boolean {
    if (!this.dataForm.errors || !this.dataForm.errors.fileError || !this.firstSubmitDone) {
      return false;
    }
    return true;
  }

  getMessageFileError(): string {
    let message = "";
    if (!this.dataForm.errors || !this.dataForm.errors.fileError || !this.firstSubmitDone) {
      return null;
    }
    if (this.dataForm.errors.fileError.empty) {
      message += "Es necesario subir el fichero que contenga la capa.";
    }

    return message;
  }

}
