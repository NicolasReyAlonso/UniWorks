import {AfterViewInit, Component, OnDestroy, OnInit, ViewChild} from '@angular/core';
import {NzModalModule, NzModalService} from 'ng-zorro-antd/modal';
import {UntypedFormGroup, UntypedFormControl, Validators, FormsModule, ReactiveFormsModule} from '@angular/forms';
import { BackendService } from 'src/app/services/backend.service';
import {MessageLogService} from 'src/app/services/message-log.service';
import {StateService} from 'src/app/services/state.service';
import { Router } from '@angular/router';
import {DynamicBrowseComponent} from '../../../../components/dynamics/dynamic-browse/dynamic-browse.component';
import {InternationalizationService} from "ngt-gui";import {AuthService} from "../../../../services/auth.service";
import IView from "../../../../interfaces/view.interfaces";
import { CommonModule } from '@angular/common';
import {NzInputModule} from "ng-zorro-antd/input";
import {NzFormModule} from "ng-zorro-antd/form";
import {SharedModule} from "../../../../shared-module/shared.module";

@Component({
    selector: 'app-organisms-browse',
    imports: [
        CommonModule,
        NzInputModule,
        NzFormModule,
        SharedModule,
        FormsModule,
        ReactiveFormsModule,
        NzModalModule,
        DynamicBrowseComponent,
    ],
    templateUrl: './organisms-browse.component.html',
    styleUrls: ['./organisms-browse.component.sass']
})
export class OrganismsBrowseComponent implements OnInit, OnDestroy, AfterViewInit, IView {

  BREADCRUMB_NAME = 'METADATA.ORGANISMS.BROWSER.BREADCRUMB';

  loading = true;
  @ViewChild(DynamicBrowseComponent) dynamicBrowser: DynamicBrowseComponent;

  showForm = false;
  dataForm: UntypedFormGroup = new UntypedFormGroup({
    genus: new UntypedFormControl(null, Validators.required),
    species: new UntypedFormControl(null, Validators.required),
    common_name: new UntypedFormControl(),
    abbreviation: new UntypedFormControl(),
    comment: new UntypedFormControl(),
  });

  constructor(
    private backendService: BackendService,
    private logService: MessageLogService,
    private modal: NzModalService,
    private stateService: StateService,
    private router: Router,
    private readonly internationalizationService: InternationalizationService,
    public readonly authService: AuthService,
  ) {}

  ngOnInit(): void {

  }

  async ngAfterViewInit(): Promise<void> {
    this.loading = true;
    const getState = await this.stateService.getStateCurrenView();
    if (getState) {
      const dynamicTable = this.dynamicBrowser.dynamicTable
      if (getState.pageIndex) {
        dynamicTable.pageIndex = getState.pageIndex;
      }
      if (getState.pageSize) {
        dynamicTable.pageSize = getState.pageSize;
      }
      if (getState.checkedItems && getState.checkedItems instanceof Array) {
        dynamicTable.checkedItems = new Set(getState.checkedItems);
      }
      this.dynamicBrowser.changePage();
    }
    this.loading = false;
  }

  ngOnDestroy(): void {
    const dynamicTable = this.dynamicBrowser.dynamicTable;
    this.stateService.setStateCurrentView({
      pageIndex: dynamicTable.pageIndex,
      pageSize: dynamicTable.pageSize,
      checkedItems: Array.from(dynamicTable.checkedItems),
    });
  }

  showModal(): void {
    this.showForm = true;
  }

  handleOk(): void {
    this.createOrganism();
  }

  handleCancel(): void {
    this.showForm = false;
  }

  createOrganism() {
    this.loading = true;
    this.backendService.postOrganisms(this.dataForm.value)
    .subscribe(
      {next: (response: any) => {
        this.logService.addIssues(response.issues);
      }, error: error => {
        this.loading = false;
        this.logService.addIssues(error.issues);
      }, complete: () => {
        this.loading = false;
        this.showForm = false;
        this.dynamicBrowser.changePage();
      }});
  }

  item2str(item): string {
    return JSON.stringify(item);
  }

  // OPEN
  openItem(item) {
    this.router.navigate(['organismDetail', item.organism_id]);
  }

  // EXPORT
  exportSelection() {
  }

  exportItem(item) {
    this.logService.info('DUMMY: Exportando la secuencia ' + this.item2str(item));
  }

  // DELETE
  async deleteSelection(event) {
    if (!event.filter.organism_id) { return; }
    const organismIds = event.filter.organism_id.unary;
    this.modal.confirm({
      nzTitle: `<i>${await this.internationalizationService.translate('METADATA.ORGANISMS.BROWSER.DELETE_SELECTION_MODAL.TITLE')}</i>`,
      nzContent: `<b>${await this.internationalizationService.translate('METADATA.ORGANISMS.BROWSER.DELETE_SELECTION_MODAL.CONTENT')} </b> ${organismIds.length}`,
      nzCentered: true,
      nzOnOk: () => {
        this.backendService.deleteOrganisms(undefined, { id: organismIds }).subscribe(
          {next: (response: any) => { this.logService.addIssues(response.issues); },
            error: error => { this.logService.addIssues(error.issues); },
          complete: () => { this.dynamicBrowser.changePage(); }}
        );
      }
    });
  }

  async deleteItem(event) {
    this.loading = true;
    this.modal.confirm({
      nzTitle: `<i>${await this.internationalizationService.translate('METADATA.ORGANISMS.BROWSER.DELETE_ITEM_MODAL.TITLE')}</i>`,
      nzContent: this.item2str(event),
      nzCentered: true,
      nzOnOk: () => {
        this.loading = true;
        this.backendService.deleteOrganisms(event.organism_id).subscribe(
          {next: (response: any) => {
            this.logService.addIssues(response.issues);
          }, error: error => {
            this.loading = false;
            this.logService.addIssues(error.issues);
          }, complete: () => {
            this.loading = false;
            this.dynamicBrowser.changePage();
          }});
        }
      });
    this.loading = false;
  }
}
