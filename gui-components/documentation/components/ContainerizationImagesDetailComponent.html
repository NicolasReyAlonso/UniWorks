import {Component, DestroyRef, inject, OnInit} from '@angular/core';
import {CommonModule} from '@angular/common';
import {BackButtonComponent} from "@components/miscellaneous/back-button/back-button.component";
import {NzPageHeaderModule} from "ng-zorro-antd/page-header";
import {TranslateModule} from "@ngx-translate/core";
import {NzTabsModule} from "ng-zorro-antd/tabs";
import {NzButtonModule} from "ng-zorro-antd/button";
import {
  ContainerizationImagesDetailGeneralTabComponent
} from "@components/admin/containerization-images/containerization-images-detail-general-tab/containerization-images-detail-general-tab.component";
import {BehaviorSubject} from "rxjs";
import {AbstractControl, FormGroup, NonNullableFormBuilder} from "@angular/forms";
import {takeUntilDestroyed} from "@angular/core/rxjs-interop";
import {BackendService} from "@services/backend.service";
import {ActivatedRoute} from "@angular/router";
import {ModalInfoService} from "@services/modal-info.service";
import {ContainerizationImageItem} from "@models/containerization-images.model";
import {NotificationService, TypeNotificationEnum} from "@services/notification.service";
import {MessageLogService} from "@services/message-log.service";

@Component({
    selector: 'app-containerization-images-detail',
    imports: [
        CommonModule,
        BackButtonComponent,
        NzPageHeaderModule,
        TranslateModule,
        NzTabsModule,
        NzButtonModule,
        ContainerizationImagesDetailGeneralTabComponent
    ],
    templateUrl: './containerization-images-detail.component.html',
    styleUrls: ['./containerization-images-detail.component.sass']
})
export class ContainerizationImagesDetailComponent implements OnInit {

  private readonly _fb: NonNullableFormBuilder = inject(NonNullableFormBuilder)
  private readonly _destroyRef: DestroyRef = inject(DestroyRef)
  private readonly _backend: BackendService = inject(BackendService)
  private readonly _activatedRoute: ActivatedRoute = inject(ActivatedRoute)
  private readonly _modalInfo: ModalInfoService = inject(ModalInfoService)
  private readonly _messageLog: MessageLogService = inject(MessageLogService)
  private readonly _notification: NotificationService = inject(NotificationService)

  private id: string;

  protected displayName: string = ''
  protected formChange: boolean
  protected formGroup: FormGroup = this._fb.group({
    name: ['', [this.emptyValidationError]],
    imageType: ['', [this.emptyValidationError]],
    location: [''],
    recipe: ['{}', this.jsonValidator]
  })
  protected readonly loading: BehaviorSubject<boolean> = new BehaviorSubject(true)
  protected containerizationImageDetailOriginalData: BehaviorSubject<ContainerizationImageItem> = new BehaviorSubject(null)

  ngOnInit(): void {
    this.id = this._activatedRoute.snapshot.params.id
    this.loadData()
    this.formGroup.valueChanges.pipe(takeUntilDestroyed(this._destroyRef)).subscribe({
      next: () => {
        this.formChange = true
      }
    })
  }

  private loadData(): void {
    this.loading.next(true)
    const getRequest = this._backend.getContainerizationImageById(this.id).subscribe({
      next: (response) => {
        getRequest.unsubscribe()
        this.containerizationImageDetailOriginalData.next(response.content)
        this.displayName = this.containerizationImageDetailOriginalData.value.name
        this.formChange = false
        this.loading.next(false)
      },
      error: (error) => {
        getRequest.unsubscribe()
        this._modalInfo.showModalInfoDefaultError(error)
        this.loading.next(false)
      }
    })

  }

  protected cancel(): void {
    this.loadData()
  }

  protected onClickModify(): void {
    if (!this.formGroup.valid) return

    this.loading.next(true)
    const requestBody: Partial<ContainerizationImageItem> = {
      name: this.formGroup.get('name').value,
      image_type: this.formGroup.get('imageType').value,
      location: this.formGroup.get('location').value,
      recipe: this.formGroup.get('recipe').value,
    }

    const putRequest = this._backend.putContainerizationImageById(this.id, requestBody).subscribe({
      next: (response: any) => {
        this._messageLog.addIssues(response.issues)
        this._notification.createNotificationWithType(
          TypeNotificationEnum.success,
          'PROCESSES_ADMIN.CONTAINERIZATION_IMAGES.NOTIFICATION_UPDATE_SUCCESS.TITLE',
          'PROCESSES_ADMIN.CONTAINERIZATION_IMAGES.NOTIFICATION_UPDATE_SUCCESS.CONTENT',
          'bottomRight'
        )
        this.loading.next(false)
        putRequest.unsubscribe()
        this.loadData()
      },
      error: (e) => {
        putRequest.unsubscribe()
        this._messageLog.addResponseIssues(e)
        this._modalInfo.showModalInfoDefaultError(e)
        this._notification.createNotificationError('PROCESSES_ADMIN.CONTAINERIZATION_IMAGES.NOTIFICATION_UPDATE_ERROR.CONTENT', 'bottomRight').then()
        this.loading.next(false)
      }
    })
  }

  private emptyValidationError(control: AbstractControl): { [key: string]: boolean } {
    const value = control.value

    if (!value || value === '') {
      return { EMPTY_ERROR: true }
    }

    return null
  }

  private jsonValidator (control: AbstractControl): { [key: string]: any } {
    const value = control.value as string
    try {
      JSON.parse(value)
    } catch (e) {
      if (e instanceof SyntaxError) {
        return { JSON_ERROR: true }
      }
    }

    return null
  }

}
