import {
  AfterViewChecked,
  ChangeDetectorRef,
  Component,
  ElementRef,
  EventEmitter,
  OnDestroy,
  OnInit,
  ViewChild
} from '@angular/core';
import {ActivatedRoute, Router} from '@angular/router';
import {NzModalRef, NzModalService} from 'ng-zorro-antd/modal';
import {forkJoin, lastValueFrom, Subject} from 'rxjs';
import {DynamicBrowseComponent} from 'src/app/components/dynamics/dynamic-browse/dynamic-browse.component';
import {AuthService} from 'src/app/services/auth.service';
import {BackendService} from 'src/app/services/backend.service';
import {MessageLogService} from 'src/app/services/message-log.service';
import {NotificationService, TypeNotificationEnum} from 'src/app/services/notification.service';
import {StateService} from '../../../../services/state.service';
import {IdentityFormComponent} from 'src/app/components/admin/identities/identity-form/identity-form.component';
import IView from '../../../../interfaces/view.interfaces';
import {CommonModule} from '@angular/common';
import {NzButtonModule} from "ng-zorro-antd/button";
import {SharedModule} from "../../../../shared-module/shared.module";

@Component({
    selector: 'app-identities-browser',
    imports: [
        CommonModule,
        IdentityFormComponent,
        DynamicBrowseComponent,
        NzButtonModule,
        SharedModule,
    ],
    templateUrl: './identities-browser.component.html',
    styleUrls: ['./identities-browser.component.sass']
})
export class IdentitiesBrowserComponent implements OnInit, OnDestroy, AfterViewChecked, IView {

  BREADCRUMB_NAME = 'SECURITY_ADMIN.IDENTITIES.BROWSER.TITLE';

  @ViewChild(DynamicBrowseComponent)
  private dynamicBrowser: DynamicBrowseComponent;

  state = {};
  buttons;

  @ViewChild('containerPage', {static: false}) containerPage: ElementRef;

  excludeFields = ['groups', 'roles', 'organizations', 'authenticators', 'versions'];
  loading = false;

  // Create Modal
  @ViewChild('modalCreateTitle', {static: false}) modalCreateTitle;
  @ViewChild('modalCreateContent', {static: false}) modalCreateContent;
  @ViewChild('identityFormElementRef', {static: false}) identityFormElementRef: IdentityFormComponent;
  @ViewChild('modalCreateFooter', {static: false}) modalCreateFooter;
  disableEditFormSubject: Subject<boolean> = new Subject<boolean>();

  constructor(
    private readonly backendService: BackendService,
    private readonly msg: MessageLogService,
    private readonly stateService: StateService,
    private readonly router: Router,
    private readonly modalService: NzModalService,
    private readonly activeRoute: ActivatedRoute,
    private readonly cdRef: ChangeDetectorRef,
    private readonly notificationService: NotificationService,
    public readonly authService: AuthService,
  ) {
  }


  ngOnInit(): void {
    const identitiesData = this.activeRoute.snapshot.data.identities;
    for (const item of identitiesData.content) {
      if (item.creation_time) {
        item.creation_time = item.creation_time.replace('T', ' ');
      }
      if (item.deactivation_time) {
        item.deactivation_time = item.deactivation_time.replace('T', ' ');
      }
    }
    this.createCustomButtonsBrowser();
  }

  async ngAfterViewInit(): Promise<void> {
    this.loading = true;
    const getState = await this.stateService.getStateCurrenView();
    if (getState) {
      const dynamicTable = this.dynamicBrowser.dynamicTable
      if (getState.pageIndex) {
        dynamicTable.pageIndex = getState.pageIndex;
      }
      if (getState.pageSize) {
        dynamicTable.pageSize = getState.pageSize;
      }
      if (getState.checkedItems && getState.checkedItems instanceof Array) {
        dynamicTable.checkedItems = new Set(getState.checkedItems);
      }
      this.dynamicBrowser.changePage();
    }
    this.loading = false;
  }

  ngOnDestroy(): void {
    const dynamicTable = this.dynamicBrowser.dynamicTable;
    this.stateService.setStateCurrentView({
      pageIndex: dynamicTable.pageIndex,
      pageSize: dynamicTable.pageSize,
      checkedItems: Array.from(dynamicTable.checkedItems),
    });
  }

  ngAfterViewChecked() {
    this.cdRef.detectChanges();
  }

  private createCustomButtonsBrowser(): void {

    const enable = {
      title: 'SECURITY_ADMIN.IDENTITIES.BROWSER.ENABLE_BUTTON',
      needCheckedItem: true,
      eventEmitter: new EventEmitter(),
      cantCheckedAllItem: true,
      funcCondition: () => {
        return this.authService.havePermission('gui-users-authorize');
      }
    };

    const disable = {
      title: 'SECURITY_ADMIN.IDENTITIES.BROWSER.DISABLE_BUTTON',
      needCheckedItem: true,
      eventEmitter: new EventEmitter(),
      cantCheckedAllItem: true,
      funcCondition: () => {
        return this.authService.havePermission('gui-users-authorize');
      }
    };

    enable.eventEmitter.subscribe({next:async (event) => {
      this.loading = true;
      try {
        const ids = event.id.unary;
        const observers = {};
        for (const id of ids) {
          observers[`${id}`] = this.backendService.putIdentity(id, {can_login: true});
        }
        await lastValueFrom(forkJoin(observers), { defaultValue: undefined });
        this.notificationService.createNotificationWithType(TypeNotificationEnum.success, 'Habilitar usuarios', 'Se han habilitado los usuarios correctamente.', 'bottomRight');
      } catch (err) {
        this.notificationService.createNotificationWithType(TypeNotificationEnum.error, 'Error', 'Ha ocurrido un error al intentar habilitar los usuarios.', 'bottomRight');
      }
      this.dynamicBrowser.changePage();
      this.loading = false;
    }});

    disable.eventEmitter.subscribe({next:async (event) => {
      this.loading = true;
      try {
        const ids = event.id.unary;
        const observers = {};
        for (const id of ids) {
          observers[`${id}`] = this.backendService.putIdentity(id, {can_login: false});
        }
        await lastValueFrom(forkJoin(observers), { defaultValue: undefined });
        this.notificationService.createNotificationWithType(TypeNotificationEnum.success, 'Deshabilitar usuarios', 'Se han deshabilitado los usuarios correctamente.', 'bottomRight');
      } catch (err) {
        this.notificationService.createNotificationWithType(TypeNotificationEnum.error, 'Error', 'Ha ocurrido un error al intentar deshabilitar los usuarios.', 'bottomRight');
      }
      this.dynamicBrowser.changePage();
      this.loading = false;
    }});

    // Add list
    this.buttons = [
      enable,
      disable,
    ];
  }

  onCreateButtonClick() {
    this.loading = true;
    this.openCreateIdentityModal();
    this.loading = false;
  }

  openItem(item) {
    this.router.navigate(['identityDetail', item.id]);
  }

  modifyItem(item) {
    this.router.navigate(['identityModify', item.id]);
  }

  deleteItem(item) {
    this.modalService.confirm({
      nzTitle: '¿Seguro que desea eliminar este usuario?',
      nzContent: item.name,
      nzOkText: 'Eliminar',
      nzOkType: 'primary',
      nzOkDanger: true,
      nzOnOk: () => this.deleteIdentity(item.id),
      nzCancelText: 'Cancelar',
    });
  }

  deleteIdentity(id) {
    this.loading = true;
    this.backendService.deleteIdentity(id).subscribe({next:(response) => {
      this.dynamicBrowser.changePage();
      this.loading = false;
    }, error: (error) => {
      this.loading = false;
      console.log(error);
    }});
  }

  openCreateIdentityModal() {
    this.loading = false;
    const modal = this.modalService.create({
      nzTitle: this.modalCreateTitle,
      nzContent: this.modalCreateContent,
      nzFooter: this.modalCreateFooter,
      nzMaskClosable: false,
    });
  }

  onClickCreateIdentityModal() {
    this.identityFormElementRef.createIdentity();
  }

  disableCreateButtonIdentityModal() {
    if (!this.identityFormElementRef) {
      return true;
    } else {
      return !this.identityFormElementRef.formGroup.valid;
    }
  }

  requestEndCreateIdentity(event: { methodRequest: 'PUT' | 'POST', responseType: 'error' | 'success', data?: any }) {
    if (event.methodRequest == 'POST') {
      if (event.responseType == 'success') {
        this.modalService.closeAll();
        this.dynamicBrowser.changePage();
      }
    }
  }

  onClickDestroyModal(modalRef: NzModalRef) {
    modalRef.destroy();
  }

  goIdentityModify() {

  }

}
