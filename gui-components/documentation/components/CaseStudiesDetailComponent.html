import {AfterViewInit, Component, OnDestroy, OnInit, ViewChild} from '@angular/core';
import {FormsModule, ReactiveFormsModule, UntypedFormBuilder, UntypedFormGroup} from '@angular/forms';
import {ActivatedRoute, Router} from '@angular/router';
import {NzModalService} from 'ng-zorro-antd/modal';
import {forkJoin, lastValueFrom} from 'rxjs';
import {AuthService} from 'src/app/services/auth.service';
import {BackendService} from 'src/app/services/backend.service';
import {MessageLogService} from 'src/app/services/message-log.service';
import {NotificationService, TypeNotificationEnum} from 'src/app/services/notification.service';
import {StateService} from 'src/app/services/state.service';
import IView from '@interfaces/view.interfaces';
import {AnnotationsFormComponent} from "src/app/components/data/annotations/annotations-form/annotations-form.component";
import { CommonModule } from '@angular/common';
import {NzButtonModule} from "ng-zorro-antd/button";
import {SharedModule} from "../../../shared-module/shared.module";
import {NzTabsModule} from "ng-zorro-antd/tabs";
import {NzListModule} from "ng-zorro-antd/list";
import {NzTableModule} from "ng-zorro-antd/table";
import {DisableControlDirective} from "../../../directives/disable-control.directive";
import {NzFormModule} from "ng-zorro-antd/form";
import {NzSpinModule} from "ng-zorro-antd/spin";
import {BackButtonComponent} from "../../../components/miscellaneous/back-button/back-button.component";
import {NzPageHeaderModule} from "ng-zorro-antd/page-header";
import {NzInputModule} from "ng-zorro-antd/input";
import {NzIconModule} from "ng-zorro-antd/icon";
import {TopAreaService} from "../../../services/top-area.service";
import {
  DynamicEntitiesTableComponent
} from "../../../components/dynamics/dynamic-entities-table/dynamic-entities-table.component";

@Component({
    selector: 'app-case-studies-detail',
    imports: [
        CommonModule,
        NzButtonModule,
        AnnotationsFormComponent,
        SharedModule,
        NzTabsModule,
        NzListModule,
        NzTableModule,
        DisableControlDirective,
        NzFormModule,
        ReactiveFormsModule,
        FormsModule,
        NzSpinModule,
        BackButtonComponent,
        NzPageHeaderModule,
        NzInputModule,
        NzIconModule,
        DynamicEntitiesTableComponent,
    ],
    templateUrl: './case-studies-detail.component.html',
    styleUrls: ['./case-studies-detail.component.sass']
})
export class CaseStudiesDetailComponent implements OnInit, OnDestroy, AfterViewInit, IView {

  @ViewChild('annotationsForm', {static: false}) annotationsForm: AnnotationsFormComponent;
  @ViewChild(DynamicEntitiesTableComponent) dynamicEntitiesTableComponent: DynamicEntitiesTableComponent;

  caseStudy;
  loadingGeneral = false;
  formGroup: UntypedFormGroup;
  formGroupChange = false;
  selectedIndexTab = 0;
  readonly headerEntities = [
    {
      translation: 'STUDY_CASES.DETAIL.INPUTS.ENTITIES.TH.NAME',
      variableName: 'name'
    },
  ];
  BREADCRUMB_NAME = {
    key: 'STUDY_CASES.DETAIL.BREADCRUMB',
    params: {
      name: '',
    }
  };

  constructor(
    private readonly activeRoute: ActivatedRoute,
    private readonly backendService: BackendService,
    private readonly fb: UntypedFormBuilder,
    private readonly router: Router,
    private readonly messageLogService: MessageLogService,
    private readonly nzModalService: NzModalService,
    private readonly notificationService: NotificationService,
    public readonly authService: AuthService,
    private readonly stateService: StateService,
    private readonly topAreaService: TopAreaService,
  ) {

  }

  async ngOnInit(): Promise<void> {
    this.loadingGeneral = true;
    this.formGroup = this.fb.group({
      name: '',
    });
    this.caseStudy = this.activeRoute.snapshot.data.caseStudy;
    this.formGroup.valueChanges.subscribe({next:(change) => this.formValueChange(change)});
    const getState = await this.stateService.getStateCurrenView();
    if (!getState || !getState.formGroupChange) {
      await this.loadData(this.caseStudy);
      this.formGroupChange = false;
    } else {
      if (getState.name) {
        this.formGroup.get('name').setValue(getState.name);
      }
      if (getState.formGroupChange) {
        this.formGroupChange = getState.formGroupChange;
      }
    }
    this.loadingGeneral = false;
  }

  async ngAfterViewInit(): Promise<void> {
    this.annotationsForm.object_uuid = this.caseStudy.uuid;
    this.annotationsForm.loadAnnotations();
  }

  async ngOnDestroy(): Promise<void> {
    this.BREADCRUMB_NAME.params.name = this.formGroup.get('name').value;
    if (!this.formGroupChange) {
      this.stateService.setStateCurrentView({
        formGroupChange: this.formGroupChange,
      }).then();
    } else {
      this.stateService.setStateCurrentView({
        name: this.formGroup.get('name').value,
        formGroupChange: this.formGroupChange,
      }).then();
    }
  }

  formValueChange(change) {
    this.formGroupChange = true;
  }

  entitiesListOfDeleteChange(event) {
    this.formValueChange(null);
  }

  private async loadData(caseStudy): Promise<void> {
    this.loadingGeneral = true;
    this.formGroupChange = false;
    this.formGroup.get('name').setValue(caseStudy.name);
  }

  async modifyCaseStudy(): Promise<void> {
    this.loadingGeneral = true;
    const value = {
      name: this.formGroup.get('name').value,
    };
    try {
      const putCaseStudyResponse: any = await lastValueFrom(this.backendService.putCaseStudy(this.caseStudy.id, value), { defaultValue: undefined });
      this.topAreaService.topAreaEventEmitter.emit({type: 'load-case-studies', value: null});
      const putCaseStudyContent = putCaseStudyResponse.content;
      this.caseStudy = putCaseStudyContent;
      const observers = [];
      this.dynamicEntitiesTableComponent.listOfDelete.forEach((element) => {
        observers.push(this.backendService.deleteCaseStudyItemByCaseStudyAndItem(this.caseStudy.id, element));
      });
      if (observers.length !== 0) await lastValueFrom(forkJoin(observers));
      this.dynamicEntitiesTableComponent.listOfDelete = [];
      this.dynamicEntitiesTableComponent.loadData();
      await this.loadData(this.caseStudy);
      this.messageLogService.addIssues(putCaseStudyResponse.issues);
      this.notificationService.createNotificationWithType(
        TypeNotificationEnum.success,
        'Caso de estudio' ,
        'Se ha actualizado el caso de estudio correctamente',
        'bottomRight');
      this.loadingGeneral = false;
      this.formGroupChange = false;
    } catch (err) {
      console.error(err);
      this.notificationService.createNotificationWithType(TypeNotificationEnum.error,
        'Error', 'Hubo un error al actualizar el caso de estudio.',
        'bottomRight');
      // await this.loadData(this.caseStudy);
      this.loadingGeneral = false;
    }
  }

  async cancelCaseStudy(): Promise<void> {
    try {
      this.loadingGeneral = true;
      const caseStudyResponse: any = await lastValueFrom(this.backendService.getCaseStudies(this.caseStudy.id, {'no_attach': true, 'fast_mode': false}), { defaultValue: undefined });
      this.caseStudy = caseStudyResponse.content;
      await this.loadData(this.caseStudy);
      this.dynamicEntitiesTableComponent.listOfDelete = [];
      this.formGroupChange = false;
      this.loadingGeneral = false;
    } catch (error) {

    }
  }

  isDisableEdit() {
    return !this.authService.havePermission('gui-case-study-edit');
  }

}
