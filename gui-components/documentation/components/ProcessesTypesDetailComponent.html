import {Component, DestroyRef, inject, OnDestroy, OnInit, ViewChild} from '@angular/core';

import {NzPageHeaderModule} from "ng-zorro-antd/page-header";
import {UntypedFormGroup} from "@angular/forms";
import {SharedModule} from "src/app/shared-module/shared.module";
import {BackendService} from "@services/backend.service";
import IView from "@interfaces/view.interfaces";
import {BehaviorSubject, forkJoin, lastValueFrom, Observable, combineLatest} from "rxjs";
import {ActivatedRoute} from "@angular/router";
import {ModalInfoService} from "@services/modal-info.service";
import {NotificationService, TypeNotificationEnum} from "@services/notification.service";
import {MessageLogService} from "@services/message-log.service";

  ProcessesTypesDetailGeneralTabComponent

  ProcessesTypesDetailSourceCodeTabComponent
} from "@components/admin/processes-types/processes-types-detail-source-code-tab/processes-types-detail-source-code-tab.component";

  ProcessesTypesDetailSchemaInputTabComponent

  ProcessesTypesDetailSchemaOutputTabComponent
} from "@components/admin/processes-types/processes-types-detail-schema-output-tab/processes-types-detail-schema-output-tab.component";

@Component({
    selector: 'app-processes-types-detail',
    imports: [
        CommonModule,
        BackButtonComponent,
        NzPageHeaderModule,
        SharedModule,
        NzTabsModule,
        ProcessesTypesDetailGeneralTabComponent,
        ProcessesTypesDetailSourceCodeTabComponent,
        NzButtonModule,
        ProcessesTypesDetailAnnotationsTabComponent,
        ProcessesTypesDetailSchemaInputTabComponent,
        ProcessesTypesDetailSchemaOutputTabComponent,
    ],
    templateUrl: './processes-types-detail.component.html',
    styleUrls: ['./processes-types-detail.component.sass']
})
export class ProcessesTypesDetailComponent implements IView, OnInit, OnDestroy {

  private readonly _backend: BackendService = inject(BackendService)
  private readonly _activatedRouter: ActivatedRoute = inject(ActivatedRoute)
  private readonly _modalInfo: ModalInfoService = inject(ModalInfoService)
  private readonly _messageLog: MessageLogService = inject(MessageLogService)
  private readonly _notification: NotificationService = inject(NotificationService)
  private readonly _destroyRef = inject(DestroyRef)

  @ViewChild(ProcessesTypesDetailGeneralTabComponent) protected processesTypesDetailGeneralTabComponent: ProcessesTypesDetailGeneralTabComponent
  @ViewChild(ProcessesTypesDetailSchemaInputTabComponent) protected processesTypesDetailSchemaInputTabComponent: ProcessesTypesDetailSchemaInputTabComponent
  @ViewChild(ProcessesTypesDetailSchemaOutputTabComponent) protected processesTypesDetailSchemaOutputTabComponent: ProcessesTypesDetailSchemaOutputTabComponent

  protected displayName = ''
  protected id: any
  public readonly BREADCRUMB_NAME = {
    key: 'PROCESSES_ADMIN.PROCESS_TYPES.DETAIL.BREADCRUMB',
    params: {name: ''}
  }
  protected formChange: boolean = false
  protected schemaValuesValid = true

  protected readonly loading: BehaviorSubject<boolean> = new BehaviorSubject(true)
  protected readonly proccessOriginalData: BehaviorSubject<GetProcessResponse | null> = new BehaviorSubject(null);
  protected readonly isReadOnly: BehaviorSubject<boolean> = new BehaviorSubject(false)
  protected readonly resourcesOriginalData: BehaviorSubject<GetResourcesResponse | null> = new BehaviorSubject(null)
  protected readonly selectedResourcesOriginalData: BehaviorSubject<number[]> = new BehaviorSubject([])
  protected readonly imagesOriginalData: BehaviorSubject<ContainerizationImageItem[]> = new BehaviorSubject([])
  protected readonly selectedImagesOriginalData: BehaviorSubject<number[]> = new BehaviorSubject([])
  protected readonly sourceCodeOriginalData = new BehaviorSubject<ProcessesSourceCode[] | null>(null)
  protected readonly uuid = new BehaviorSubject<string | null>(null)
  protected readonly schemaInputsValue = new BehaviorSubject<{ value: any, isValid: boolean }>({value: null, isValid: true})
  protected readonly schemaOutputsValue = new BehaviorSubject<{ value: any, isValid: boolean }>({value: null, isValid: true})

  ngOnInit(): void {
    this.id = this._activatedRouter.snapshot.params.id;
    this.schemaValuesChange()
    this.loadData()
  }

  ngOnDestroy(): void {
    this.removeSubjects()
  }

  private schemaValuesChange(): void {
    combineLatest([
      this.schemaInputsValue.asObservable(),
      this.schemaOutputsValue.asObservable()
    ]).pipe(takeUntilDestroyed(this._destroyRef)).subscribe(([schemaInput, schemaOutputs]) => {
      this.schemaValuesValid = schemaInput.isValid && schemaOutputs.isValid;
      this.formChange = true
    })
  }

  private removeSubjects(): void {
    this.proccessOriginalData.complete()
    this.isReadOnly.complete()
    this.loading.complete()
    this.resourcesOriginalData.complete()
    this.selectedResourcesOriginalData.complete()
    this.sourceCodeOriginalData.complete()
    this.uuid.complete()
    this.schemaInputsValue.complete()
  }

  private async loadDataNext({processesResponse, resourcesResponse, imagesResponse}: {processesResponse: GetProcessResponse, resourcesResponse: GetResourcesResponse, imagesResponse: ContainerizationImagesItemResponse}): Promise<void> {
    this._messageLog.addIssues(processesResponse.issues)
    this.displayName = processesResponse.content.name
    this.proccessOriginalData.next(processesResponse)
    this.schemaInputsValue.next({value: processesResponse.content.schema_inputs, isValid: true})
    this.schemaOutputsValue.next({value: processesResponse.content.schema_outputs, isValid: true})
    this.processesTypesDetailSchemaInputTabComponent.inputValue = processesResponse.content.schema_inputs ? JSON.stringify(processesResponse.content.schema_inputs, null, 2) : ''
    this.processesTypesDetailSchemaOutputTabComponent.inputValue = processesResponse.content.schema_outputs ? JSON.stringify(processesResponse.content.schema_outputs, null, 2) : ''
    this.resourcesOriginalData.next(resourcesResponse)
    this.imagesOriginalData.next(imagesResponse.content)
    this.selectedImagesOriginalData.next(processesResponse.content.images.map(element => element.image_id))
    if (!processesResponse.content.execution) {
      this.isReadOnly.next(true)
    }
    if (!this.uuid.value) {
      this.uuid.next(processesResponse.content.uuid)
    }
    try {
      await this.resourcesAdaptationForkJoin({processesResponse})
      await this.sourcesForkJoin({processesResponse})
      this.formChange = false
    } catch (e) {

    }
    this.loading.next(false)
  }

  private async resourcesAdaptationForkJoin(args :{processesResponse: GetProcessResponse}) {
    const resourcesAdaptationForkJoinJson: { [key: number]: Observable<GetProcessesInResourceResponse> } = args.processesResponse.content.resource_adaptations.reduce((previousValue, currentValue) => {
      previousValue[currentValue] = this._backend.getProcessesInResource(currentValue)
      return previousValue
    }, {})
    if (Object.keys(resourcesAdaptationForkJoinJson).length > 0) {
      try {
        const getProcessesInResourceResponses = await lastValueFrom(forkJoin(
          resourcesAdaptationForkJoinJson
        ))
        const selectedResourcesOriginalData: number[] = []
        for (const response of Object.values(getProcessesInResourceResponses)) {
          selectedResourcesOriginalData.push(response.content.resource)
        }
        this.selectedResourcesOriginalData.next(selectedResourcesOriginalData)
      } catch (e) {
        this._messageLog.addResponseIssues(e)
        this._modalInfo.showModalInfoDefaultError(e)
        throw new Error()
      }
    } else {
      this.selectedResourcesOriginalData.next([])
    }
  }

  private async sourcesForkJoin(args :{processesResponse: GetProcessResponse}): Promise<void> {
    const sourcesForkJoin = args.processesResponse.content.sources.reduce((previousValue, currentValue) => {
      previousValue[currentValue] = this._backend.getProcessesSourceCodeById(currentValue)
      return previousValue
    }, {})
    if (Object.keys(sourcesForkJoin).length > 0) {
      try {
        const sourcesForkJoinResponses = await lastValueFrom(forkJoin(
          sourcesForkJoin
        ))
        const newSourceCodeOriginalData: ProcessesSourceCode[] = [];
        for (const response of Object.values(sourcesForkJoinResponses) as ProcessesSourceCodeResponse[]) {
          newSourceCodeOriginalData.push(response.content)
        }
        this.sourceCodeOriginalData.next(newSourceCodeOriginalData)
      } catch (e) {
        this._messageLog.addResponseIssues(e)
        this._modalInfo.showModalInfoDefaultError(e)
        throw new Error()
      }
    } else {
      this.sourceCodeOriginalData.next([])
    }
  }

  private loadData(): void {
    this.loading.next(true)
    forkJoin({
      processesResponse: this._backend.getProcess(this.id),
      resourcesResponse: this._backend.getResources(),
      imagesResponse: this._backend.getContainerizationImages()
    }).subscribe({
      next: (response) => this.loadDataNext(response),
      error: (e) => {
        this._messageLog.addResponseIssues(e)
        this._modalInfo.showModalInfoDefaultError(e)
        this.loading.next(false)
      }
    })
  }

  protected onFormChangeGeneralTab(): void {
    this.formChange = true
  }

  protected async cancel(): Promise<void> {
    this.loadData()
  }

  private modifySelectedResources(): { [key: string]: Observable<any> } {
    const generalTabFormGroup = this.processesTypesDetailGeneralTabComponent.formGroup
    const selectedResources: number[] = generalTabFormGroup.get('resources').value
    const selectedResourcesOriginalData: number[] = this.selectedResourcesOriginalData.value

    const addSelectedResources: number[] = []
    const deleteSelectedResources: number[] = []

    for (const resourceId of selectedResources) {
      if (!selectedResourcesOriginalData.includes(resourceId)) {
        addSelectedResources.push(resourceId)
      }
    }

    for (const resourceId of selectedResourcesOriginalData) {
      if (!selectedResources.includes(resourceId)) {
        deleteSelectedResources.push(resourceId)
      }
    }

    return {
      ...addSelectedResources.reduce((previousValue, currentValue) => {
        previousValue[`addSelectedResources-${currentValue}`] = this._backend.postProcessesInResource({"process_id": `${this.id}`,"resource_id": `${currentValue}`})
        return previousValue
      }, {}),
      ...deleteSelectedResources.reduce((previousValue, currentValue) => {
        previousValue[`deleteSelectedResources-${currentValue}`] = this._backend.deleteProcessesInResource(currentValue, this.id)
        return previousValue
      }, {}),
    }
  }

  private modifySelectedImages(): { [key: string]: Observable<any> } {
    const generalTabFormGroup = this.processesTypesDetailGeneralTabComponent.formGroup
    const selectedImages: number[] = generalTabFormGroup.get('images').value
    const selectedImagesOriginalData: number[] = this.selectedImagesOriginalData.value

    const addSelectedImages: number[] = []
    const deleteSelectedImages: number[] = []

    for (const imageId of selectedImages) {
      if (!selectedImagesOriginalData.includes(imageId)) {
        addSelectedImages.push(imageId)
      }
    }

    for (const imageId of selectedImagesOriginalData) {
      if (!selectedImages.includes(imageId)) {
        deleteSelectedImages.push(imageId)
      }
    }

    return {
      ...addSelectedImages.reduce((previousValue, currentValue) => {
        previousValue[`addSelectedImages-${currentValue}`] = this._backend.postProcessImages({process_id: this.id, image_id: currentValue})
        return previousValue
      }, {}),
      ...deleteSelectedImages.reduce((previousValue, currentValue) => {
        previousValue[`deleteSelectedImagess-${currentValue}`] = this._backend.deleteProcessImages(this.id, currentValue)
        return previousValue
      }, {})
    }
  }

  private putProcessByForm() {
    const formGeneralTabValue = this.processesTypesDetailGeneralTabComponent.formGroup.value

    const body = {
      ...this.proccessOriginalData.value.content
    }

    delete body.sources
    delete body.images
    delete body.algorithms
    delete body.resource_adaptations
    delete body.resource_jm_adaptations
    delete body.files
    delete body.collections
    delete body.case_studies
    delete body.annotations
    delete body.related_objects
    delete body.related_subjects

    body.name = formGeneralTabValue.name
    body.description = formGeneralTabValue.description
    if (body.execution) {
      body.execution.kernel = formGeneralTabValue.kernel
      body.execution.geoprocess_type = formGeneralTabValue.geoprocessType
      body.execution.system = formGeneralTabValue.system
    }
    body.schema_inputs = this.schemaInputsValue.value.value
    body.schema_outputs = this.schemaOutputsValue.value.value

    return this._backend.putProcess(this.id, body)
  }

  protected onClickModify(): void {

    let queries: {[key: string | number]: Observable<any>} = {}

    if (this.formChange && this.processesTypesDetailGeneralTabComponent.formGroup.valid && this.schemaValuesValid) {
      queries = {...queries, ...this.modifySelectedResources(), ...this.modifySelectedImages()}
    }

    queries['putProcess'] = this.putProcessByForm()

    if (Object.keys(queries).length > 0) {
      this.loading.next(true)
      forkJoin(queries).subscribe({
        next: (responses) => {
          for (const response of Object.values(responses)) {
            this._messageLog.addIssues(response.issues)
          }
          this._notification.createNotificationWithType(
              TypeNotificationEnum.success,
              'PROCESSES_ADMIN.PROCESS_TYPES.DETAIL.NOTIFICATION_UPDATE_SUCCESS.TITLE',
              'PROCESSES_ADMIN.PROCESS_TYPES.DETAIL.NOTIFICATION_UPDATE_SUCCESS.CONTENT',
              'bottomRight'
          )
          this.loadData()
        },
        error: (e) => {
          this._messageLog.addResponseIssues(e)
          this._modalInfo.showModalInfoDefaultError(e)
          this._notification.createNotificationError('PROCESSES_ADMIN.PROCESS_TYPES.DETAIL.NOTIFICATION_UPDATE_ERROR.CONTENT', 'bottomRight')
          this.loading.next(false)
        }
      })
    }
  }
}
