import {AfterViewChecked, Component, Input, OnDestroy, OnInit, ViewChild} from '@angular/core';
import { ActivatedRoute, Router } from '@angular/router';
import {lastValueFrom, Subject} from "rxjs";
import {BackendService} from "src/app/services/backend.service";
import {MessageLogService} from "src/app/services/message-log.service";
import { CommonModule } from '@angular/common';
import {
  ArchaeopteryxBrowserComponent
} from "../../../components/display/archaeopteryx/archaeopteryx-browser/archaeopteryx-browser.component";
import {NzCardModule} from "ng-zorro-antd/card";
import {SharedModule} from "../../../shared-module/shared.module";
import {NzDividerModule} from "ng-zorro-antd/divider";
import {NzSelectModule} from "ng-zorro-antd/select";
import {FormsModule, ReactiveFormsModule} from "@angular/forms";
import {ColorPickerModule} from "ngx-color-picker";
import {NzPageHeaderModule} from "ng-zorro-antd/page-header";
import {NexusTreeSelectorComponent} from "../../../components/miscellaneous/nexus-tree-selector/nexus-tree-selector.component";
import {NzButtonModule} from "ng-zorro-antd/button";
import {NzIconModule} from "ng-zorro-antd/icon";

@Component({
    selector: 'app-archaeopteryx-page',
    imports: [
        CommonModule,
        ArchaeopteryxBrowserComponent,
        NzCardModule,
        SharedModule,
        NzDividerModule,
        NzSelectModule,
        FormsModule,
        ReactiveFormsModule,
        ColorPickerModule,
        NzPageHeaderModule,
        NexusTreeSelectorComponent,
        NzButtonModule,
        NzIconModule
    ],
    templateUrl: './archaeopteryx-page.component.html',
    styleUrls: ['./archaeopteryx-page.component.sass']
})
export class ArchaeopteryxPageComponent implements OnInit, AfterViewChecked, OnDestroy {

  url: string;
  @Input() tree: string;
  view = null;

  @ViewChild('archaeopteryxBrowserComponent', { static: false}) archaeopteryxBrowserComponent: ArchaeopteryxBrowserComponent;

  BREADCRUMB_NAME = {
    key: 'MOLECULAR_DATA.PHYLOTREES.ARCHAEOPTERYX.BREADCRUMB',
    params: {}
  };
  constructor(
    private readonly route: ActivatedRoute,
    private readonly router: Router,
    private readonly backendService: BackendService,
    private readonly msg: MessageLogService
  ) {
  }

  async ngOnInit() {
    if (!this.route.snapshot.queryParams['url'] && !this.route.snapshot.queryParams['view_id']) {
      await this.router.navigate(['home']);
    }
    if (this.route.snapshot.queryParams['view_id']) {
      const response = await lastValueFrom(this.backendService.getViewers(this.route.snapshot.queryParams['view_id']), { defaultValue: undefined });
      if (response['issues'].length !== 0) {
        this.msg.addIssues(response['issues']);
      }
      this.view = response['content'];
      this.url = this.view['data']['filesApiUrl'];
      const waitForExist = setInterval(() => {
        if (this.archaeopteryxBrowserComponent.savedColors) {
          this.archaeopteryxBrowserComponent.savedColors = this.view['data']['savedColors'];
          clearInterval(waitForExist);
        }
      }, 300);
    } else {
      this.url = this.route.snapshot.queryParams['url'];
    }
  }

  receiveTree(tree): void {
   this.tree = tree.newick;
  }

  ngAfterViewChecked() {
    Array.from(document.getElementsByTagName('nz-option-item')).forEach((element) => {
      element.setAttribute('style', 'color: ' + element.getAttribute('title'));
    });
  }

  ngOnDestroy() {

  }
}
