import {Component, OnInit} from '@angular/core';
import {CommonModule} from '@angular/common';
import {BackButtonComponent} from "../../../../components/miscellaneous/back-button/back-button.component";
import {NzPageHeaderModule} from "ng-zorro-antd/page-header";
import {TranslateModule} from "@ngx-translate/core";
import {NzSpinModule} from "ng-zorro-antd/spin";
import {
  AbstractControl,
  FormsModule,
  ReactiveFormsModule,
  UntypedFormBuilder,
  UntypedFormGroup,
  ValidationErrors
} from "@angular/forms";
import {NzFormModule} from "ng-zorro-antd/form";
import {NzInputModule} from "ng-zorro-antd/input";
import {AuthService} from "../../../../services/auth.service";
import {BackendService} from "../../../../services/backend.service";
import {ActivatedRoute} from "@angular/router";
import {forkJoin, lastValueFrom} from "rxjs";
import {NotificationService, TypeNotificationEnum} from "../../../../services/notification.service";
import {GlobalService} from "../../../../services/global.service";
import {NzSelectModule} from "ng-zorro-antd/select";
import {NzButtonModule} from "ng-zorro-antd/button";
import {MessageLogService} from "../../../../services/message-log.service";
import IView from "../../../../interfaces/view.interfaces";

@Component({
    selector: 'app-compute-resources-detail',
    imports: [
        CommonModule,
        BackButtonComponent,
        NzPageHeaderModule,
        TranslateModule,
        NzSpinModule,
        FormsModule,
        NzFormModule,
        NzInputModule,
        ReactiveFormsModule,
        NzSelectModule,
        NzButtonModule
    ],
    templateUrl: './compute-resources-detail.component.html',
    styleUrls: ['./compute-resources-detail.component.sass']
})
export class ComputeResourcesDetailComponent implements OnInit, IView {

  id;
  displayName = "";
  loading = false;
  formGroup: UntypedFormGroup;
  formGroupChange = false;
  jmTypesOptions = [];
  processesOptions = [];
  launchable_processesBeforeUpdate;

  BREADCRUMB_NAME = {
    key: 'PROCESSES_ADMIN.COMPUTE_RESOURCES.DETAIL.BREADCRUMB',
    params: { name: '' },
  }

  constructor(
    private readonly fb: UntypedFormBuilder,
    public readonly authService: AuthService,
    private readonly backendService: BackendService,
    private readonly activatedRouter: ActivatedRoute,
    private readonly notificationService: NotificationService,
    private readonly globalService: GlobalService,
    private readonly messageLogService: MessageLogService,
  ) {
  }

  async ngOnInit(): Promise<void> {
    this.loading = true;
    const activatedRouteSnapshot = this.activatedRouter.snapshot;
    this.id = activatedRouteSnapshot.params.id;
    this.initFormGroup();
    await lastValueFrom(forkJoin([
      this.loadJmTypesOptions(),
      this.loadDataFromBackend(),
      this.loadProcessesOptions(),
    ]));
    this.loading = false;
  }

  async loadJmTypesOptions(): Promise<void> {
    try {
      const responseJmTypes: any = await lastValueFrom(this.backendService.getJmTypes());
      this.jmTypesOptions = responseJmTypes.content;
    } catch (e) {
      this.loading = false;
    }
  }

  async loadProcessesOptions(): Promise<void> {
    try {
      const response: any = await lastValueFrom(this.backendService.getProcesses());
      this.processesOptions = response.content;
    } catch (e) {
      this.loading = false;
    }
  }

  private initFormGroup(): void {
    this.formGroup = this.fb.group({
      id: [''],
      uuid: [''],
      name: [''],
      launchable_processes: [[]],
      jm_type: [null],
      jm_location: [''],
      jm_credentials: [''],
      jm_params: [''],
      agreement: ['']
    }, {
      validators: [
        this.nameValidator.bind(this),
        this.jmTypeValidator.bind(this),
        this.locationValidator.bind(this),
        this.credentialsValidator.bind(this),
        this.paramsValidator.bind(this),
        this.agreementValidator.bind(this),
      ]
    });
    this.formGroup.get('id').setValue(this.id);
    this.setDisableFields();
    this.formGroup.valueChanges.subscribe((change) => this.formValueChange(change));
  }

  formValueChange(change): void {
    this.formGroupChange = true;
  }

  private setDisableFields(): void {
    this.formGroup.get('id').disable();
    this.formGroup.get('uuid').disable();
    if (!this.authService.havePermission('gui-cresource-edit')) {
      this.formGroup.get('launchable_processes').disable();
      this.formGroup.get('name').disable();
      this.formGroup.get('jm_type').disable();
      this.formGroup.get('jm_location').disable();
      this.formGroup.get('jm_credentials').disable();
      this.formGroup.get('jm_params').disable();
      this.formGroup.get('agreement').disable();
    }
  }

  async loadDataFromBackend(): Promise<void> {
    try {
      const response: any = await lastValueFrom(this.backendService.getResourceById(this.id));
      const content = response.content;
      this.formGroup.get('uuid').setValue(content.uuid);
      this.formGroup.get('name').setValue(content.name);
      this.displayName = content.name;
      this.BREADCRUMB_NAME.params.name = this.displayName;
      this.formGroup.get('jm_type').setValue(content.jm_type);
      const launchableProcesses =  await this.loadLaunchableProcessesFromBackend(content.process_adaptations);
      this.formGroup.get('launchable_processes').setValue(launchableProcesses);
      this.launchable_processesBeforeUpdate = [...launchableProcesses];
      this.formGroup.get('jm_location').setValue(content.jm_location ? JSON.stringify(content.jm_location) : '');
      this.formGroup.get('jm_credentials').setValue(content.jm_credentials ? JSON.stringify(content.jm_credentials) : '');
      this.formGroup.get('jm_params').setValue(content.jm_params ? JSON.stringify(content.jm_params) : '');
      this.formGroup.get('agreement').setValue(content.agreement ? JSON.stringify(content.agreement) : '');
      this.formGroupChange = false;
      this.messageLogService.addIssues(response.issues);
    } catch (e) {
      this.messageLogService.addIssues(e.issues);
      this.notificationService.createNotificationWithType(
        TypeNotificationEnum.error,
        'PROCESSES_ADMIN.COMPUTE_RESOURCES.DETAIL.NOTIFICATION_LOAD_DATA_ERROR.TITLE',
        'PROCESSES_ADMIN.COMPUTE_RESOURCES.DETAIL.NOTIFICATION_LOAD_DATA_ERROR.CONTENT',
        'bottomRight'
      );
    }
  }

  async loadLaunchableProcessesFromBackend(process_adaptations): Promise<any[]> {
    if (!process_adaptations) return [];
    const observables = [];
    let launchableProcesses = [];
    for (let item of process_adaptations) {
      observables.push(this.backendService.getProcessesInResource(item));
    }
    if (observables.length > 0) {
      const response: any[] = await lastValueFrom(forkJoin(observables));
      launchableProcesses = response.map((item) => {
        return item.content.process_id;
      });
    }
    return launchableProcesses;
  }

  async cancelButtonClick(): Promise<void> {
    this.loading = true;
    await this.loadDataFromBackend();
    this.loading = false;
  }

  async update(): Promise<void> {
    this.loading = true;
    if (!this.formGroup.valid) return;
    const launchableProcessesObservables = this.createObservablesForLaunchableProcesses();
    try {
      if (launchableProcessesObservables.length > 0) {
        await lastValueFrom(forkJoin(launchableProcessesObservables));
      }
      this.launchable_processesBeforeUpdate = [...this.formGroup.get('launchable_processes').value];
      const value = {
        name: this.formGroup.get('name').value,
        jm_type: this.formGroup.get('jm_type').value,
        jm_location: this.formGroup.get('jm_location').value === '' ? null : this.globalService.convertToJson(this.formGroup.get('jm_location').value),
        jm_credentials: this.formGroup.get('jm_credentials').value === '' ? null : this.globalService.convertToJson(this.formGroup.get('jm_credentials').value),
        jm_params: this.formGroup.get('jm_params').value === '' ? null : this.globalService.convertToJson(this.formGroup.get('jm_params').value),
        agreement: this.formGroup.get('agreement').value === '' ? null : this.globalService.convertToJson(this.formGroup.get('agreement').value),
      };
      const response: any = await lastValueFrom(this.backendService.putResourceById(this.id, value));
      this.displayName = value.name;
      this.BREADCRUMB_NAME.params.name = this.displayName;
      this.messageLogService.addIssues(response.issues);
    } catch (e) {
      this.notificationService.createNotificationWithType(
        TypeNotificationEnum.error,
        'PROCESSES_ADMIN.COMPUTE_RESOURCES.DETAIL.NOTIFICATION_UPDATE_ERROR.TITLE',
        'PROCESSES_ADMIN.COMPUTE_RESOURCES.DETAIL.NOTIFICATION_UPDATE_ERROR.CONTENT',
        'bottomRight'
      );
      this.messageLogService.addIssues(e.issues);
    }
    this.loading = false;
  }

  createObservablesForLaunchableProcesses() {
    const changes: any = {};
    const observables: any[] = [];
    const newData = this.formGroup.get('launchable_processes').value;
    for (let item of newData) {
      changes[item] = {
        newData: true,
        oldData: false,
      }
    }
    for (let item of this.launchable_processesBeforeUpdate) {
      if (changes[item]) {
        changes[item].oldData = true;
      } else {
        changes[item] = {
          newData: false,
          oldData: true,
        }
      }
    }
    for (let key of Object.keys(changes)) {
      const change = changes[key];
      if (change.newData && !change.oldData) {
        observables.push(this.backendService.postProcessesInResource({process_id: key, resource_id: this.id}));
      } else if (!change.newData && change.oldData) {
        observables.push(this.backendService.deleteProcessesInResource(this.id, key));
      }
    }
    return observables;
  }

  private nameValidator(control: AbstractControl): ValidationErrors | null {
    let error = false;
    let emptyError = false;
    const name = control.get('name').value;
    if (!name || name == "") {
      error = true;
      emptyError = true;
    }
    return error ? {
      nameError: {
        empty: emptyError,
      }
    } : null;
  }

  getMessageNameError(): string[] {
    let message = [];
    if (!this.formGroup.errors || !this.formGroup.errors.nameError) {
      return message;
    }
    if (this.formGroup.errors.nameError.empty) {
      message.push('PROCESSES_ADMIN.COMPUTE_RESOURCES.DETAIL.INPUTS.NAME.EMPTY_ERROR');
    }
    return message;
  }

  jmTypeValidator(control: AbstractControl): ValidationErrors | null {
    let error = false;
    let emptyError = false;
    const jm_type = control.get('jm_type').value;
    if (!jm_type) {
      error = true;
      emptyError = true;
    }
    return error ? {
      jm_typeError: {
        empty: emptyError,
      }
    } : null;
  }

  getMessageJmTypeError(): string[] {
    let message = [];

  }

  private locationValidator(control: AbstractControl): ValidationErrors | null {
    let error = false;
    let jsonError = false;
    const location = control.get('jm_location').value;
    if (location && location !== '' && !this.globalService.convertToJson(location)) {
      error = true;
      jsonError = true;
    }
    return error ? {
      locationError: {
        json: jsonError,
      }
    } : null;
  }
