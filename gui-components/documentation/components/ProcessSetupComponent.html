export class ProcessSetupComponent extends FormlyUser implements OnInit, AfterViewChecked, IView {

  BREADCRUMB_NAME = "PROCESSES.PROCESSES_SETUP.BREADCRUMB";

  constructor(public readonly backendService: BackendService,
              public readonly msg: MessageLogService,
              private readonly route: ActivatedRoute,
              private readonly router: Router,
              private readonly changeDetector: ChangeDetectorRef,
              private readonly fb: UntypedFormBuilder,
              private readonly modal: NzModalService,
              private readonly translate: TranslateService,
              private readonly internationalizationService: InternationalizationService,
              private readonly fileService: FilesService,
  ) {
    super(fileService);
  }

  searchValue = '';
  tree = [];
  fileList = [];
  file: any;
  loading = false;
  step = 0;
  validConfiguration = true;
  expandFiles = false;
  processes: any;
  selectedProcess: number;
  resources: any;
  selectedResource: number;
  selectedPriority = 'Media';
  jobContext: any = {};
  path?: string;
  form = new UntypedFormGroup({});
  selectProgramForm!: UntypedFormGroup;
  options: FormlyFormOptions = {};
  myFilter: any = {};
  showSelectedResourceError = false;
  showSelectedProcessError = false;
  //HPC parameters
  nodes: number;
  nTasks: number;
  cpusPerTask: number;
  cpusPerGpu: number;
  gpus: number;
  priority: number;
  time: number;
  relaunchModelLoading: boolean;
  confirmModal?: NzModalRef;
  maxNodes: number;
  maxThreads: number;
  maxTime: number;
  minTasks: number;
  minCpusPerTask: number;
  maxCpusPerTask: number;
  warning: string;
  tip: string;
  processParallelizationType: string;
  math = Math;
  disableProgramFCInfoButton = false;
  loadingProgramFCInfoButton = false;
  isVisibleProgramFCInfoModal = false;
  protected needImage = false
  imagesOptions: ContainerizationImageItem[] = []

  ngOnInit(): void {
    // const organisms = this.route.snapshot.data.organisms;
    // this.tree = this.orgs2nztree(organisms);
    this.searchValue = '';
    this.processes = this.route.parent.snapshot.data.processes.content;
    let ps = [];
    for (let p of this.processes) {
      if (p.name != null) {
        ps.push(p)
      }
    }
    this.processes = ps;
    const isHPCValidator = (control: UntypedFormControl): { [s: string]: boolean } => {
      if (!control.value &&
        this.selectedResource !== undefined &&
        (this.resources?.content[this.selectedResource]?.name.endsWith('slurm') ||
          this.resources?.content[this.selectedResource]?.name.endsWith('cipres'))) {
        return {required: true};
      }
      return {};
    };

    this.selectProgramForm = this.fb.group(
      {
        imageFC: [null],
        processFC: [null, [Validators.required]],
        resourceFC: [null, [Validators.required]],
        timeFC: [null, [Validators.min(0.5), isHPCValidator]],
        nodesFC: [null, [Validators.min(1), isHPCValidator]],
        nTasksFC: [null, [Validators.min(1), isHPCValidator]],
        cpusPerTaskFC: [null, [Validators.min(1), isHPCValidator]],
      });

    this.selectProgramForm.get('processFC').valueChanges.subscribe({next: this.processFCValueChange.bind(this)});

    if ('process' in history.state && 'jobInputs' in history.state) {
      for (let i = 0; i < this.processes.length; i++) {
        if (this.processes[i].name === history.state.process) {
          this.selectedProcess = i;
          break;
        }
      }
      const jobInputs = JSON.parse(JSON.parse(history.state.jobInputs));
      /*for (const data of jobInputs.data) {
        if (data && data.formlyKey && data.index !== undefined) {
          const scriptParam = jobInputs.parameters.script_params[this.processes[this.selectedProcess].name];
          if (scriptParam[data.formlyKey] === undefined) {
            scriptParam[data.formlyKey] = [];
          }
          scriptParam[data.formlyKey][data.index] = data;
        }
      }*/
      this.model = jobInputs.parameters.script_params;
      this.loadResource(false);
      this.relaunchModelLoading = true;
    }
  }

  loadImagesByProcessId(processId: number | string) {
    const imagesRequest = this.backendService.getContainerizationImagesByProcessId(processId).subscribe({
      next: (response: any) => {
        console.log(response)
        this.imagesOptions = response.content
        imagesRequest.unsubscribe()
      }
    })
  }

  ngAfterViewChecked() {
    this.changeDetector.detectChanges();
  }

  processFCValueChange(value) {
    const selectedProcessObject = this.processes[value]
    if (!selectedProcessObject || !selectedProcessObject.execution) {
      this.needImage = false
      this.selectProgramForm.get('imageFC').setValue(null)
    } else {
      this.needImage = true
      this.loadImagesByProcessId(this.processes[value].id)
    }
    if (value === undefined || value === null) {
      this.loadingProgramFCInfoButton = false;
      this.disableProgramFCInfoButton = true;
    } else {
      this.loadingProgramFCInfoButton = false;
      this.disableProgramFCInfoButton = false;
    }
  }

  onOpenProgramFCInfoClick() {
    this.isVisibleProgramFCInfoModal = true;
  }

  onOpenInNewTabProgramFCInfoClick() {
    window.open('https://gutl.jovenclub.cu/wp-content/uploads/2013/10/El+gran+libro+de+HTML5+CSS3+y+Javascrip.pdf#page=119', '_blank').focus();
  }

  closeProgramFCInfoModal() {
    this.isVisibleProgramFCInfoModal = false;
  }

  updateResourcesInputFields(processSchema: any, resourceParams: any) {
    this.processParallelizationType = undefined;
    if ('threads' in processSchema) {
      this.processParallelizationType = 'threads';
      this.minCpusPerTask = 'minCpusPerTask' in processSchema['threads'] ? processSchema['minCpusPerTask'] : 1;
      this.maxCpusPerTask = 'maxCpusPerTask' in processSchema['threads'] ? processSchema['maxCpusPerTask'] : undefined;
    } else if ('mpi' in processSchema) {
      this.processParallelizationType = 'mpi';
      this.minTasks = 'minTasks' in processSchema['mpi'] ? processSchema['minTasks'] : 1;
    } else if ('noParallelParams' in processSchema) {
      this.processParallelizationType = 'noParallelParams'
    }
    if (this.processParallelizationType !== undefined) {
      const processParams = processSchema[this.processParallelizationType];
      this.tip = 'tip' in processParams ? this.translate.instant(processParams['tip']) : undefined;
      this.warning = 'warning' in processParams ? this.translate.instant(processParams['warning']) : undefined;
      this.maxNodes = 'max_nodes' in resourceParams ? resourceParams['max_nodes'] : undefined;
      this.maxThreads = 'max_threads' in resourceParams ? resourceParams['max_threads'] : this.maxCpusPerTask;
      this.maxTime = 'max_time' in resourceParams ? resourceParams['max_time'] : undefined;
      if (this.maxThreads) {
        this.maxThreads = Math.min(this.maxThreads, this.maxCpusPerTask)
      }
    }
  }

  loadResource(resetModel: boolean) {
    if (resetModel && !this.relaunchModelLoading) {
      this.model = {};
      this.tip = undefined;
      this.warning = undefined;
    }
    this.relaunchModelLoading = false;
    if (this.selectedProcess !== undefined) {
      this.options = this.DEBUG_JSON_FORMLY_OPTIONS ?? this.processes[this.selectedProcess]?.schema_inputs.options;
      if (this.options) {
        if (!('formState' in this.options))
          this.options['formState'] = {}
        this.options['formState']['mainModel'] = this.model;
      }
      const processes = this.DEBUG_JSON_FORMLY_FIELDS ?? this.processes[this.selectedProcess]?.schema_inputs.fields;
      // Gives a field an onChange function
      const initField = function (field) {
        if (field.props && field.props.changeExpr !== undefined) {
          field.props.changeFunc = Function('$event', 'field', field.props.changeExpr).bind(this);
        }
        if (!resetModel && field.props.initExpr !== undefined) {
          this.initializeFields(field.props.initExpr, field, this.getModelValue(field.key, this.model));
        }
        return field;
      }.bind(this);

      for (const [i, subprocess] of processes.entries()) {
        if ('advanced_parameters' === subprocess.fieldGroup.at(-1).key) {
          let newFieldGroup;
          newFieldGroup = subprocess.fieldGroup.slice(0, -1).map(initField);
          subprocess.fieldGroup.at(-1).fieldGroup = subprocess.fieldGroup.at(-1).fieldGroup.map(initField)
          newFieldGroup.push(subprocess.fieldGroup.at(-1));
          processes[i].fieldGroup = newFieldGroup;
        } else {
          processes[i].fieldGroup = subprocess.fieldGroup.map(initField);
        }
      }
      this.fields = processes;
      this.backendService.getResourcesByProcessId(this.processes[this.selectedProcess].id).subscribe(
        {next: response => {
          this.selectedResource = undefined;
          this.resources = response;
          if ('resource' in history.state) {
            const resourcesContent = this.resources?.content;
            for (let i = 0; i < resourcesContent.length; i++) {
              if (resourcesContent[i].name === history.state.resource) {
                this.selectedResource = i;
              }
            }
          }
        },
        error: error => {
          this.msg.error(error.error);
          this.loading = false;
        }, complete: () => {
          this.searchValue = '';
          this.loading = false;
        }}
      );
    }
  }

  selectResource(event) {
    if (event !== undefined)
      this.updateResourcesInputFields(this.processes[this.selectedProcess]['schema_inputs'], this.resources.content[event]['jm_params']);

  private orgs2nztree(orgs) {
    const res = [];
    let tmp = {};
    for (const org of orgs) {
      tmp = {
        title: org.genus + ' ' + org.species, key: org.organism_id,
        children: []
      };
      res.push(tmp);
    }

  }

  private seqs2nztree(seqs) {
    const res = [];
    let tmp = {};
    for (const seq of seqs) {
      tmp = {
        title: seq.name, key: seq.feature_id,
        isLeaf: true
      };
      res.push(tmp);
    }
    return res;
  }

  nzEvent(event: NzFormatEmitEvent): void {
    // load child async
    if (event.eventName === 'expand') {
      const node = event.node;
      if (node?.getChildren().length === 0 && node?.isExpanded) {
        this.loadNode(node).then(data => {
          node.addChildren(data);
        });
      }
    }
  }

  loadNode(org): Promise<NzTreeNodeOptions[]> {

          this.msg.addIssues(error.issues);

        complete: () => {

        }});
    });
  }
