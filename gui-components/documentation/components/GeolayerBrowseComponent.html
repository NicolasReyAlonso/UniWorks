import {AfterViewInit, Component, EventEmitter, OnDestroy, OnInit, ViewChild} from '@angular/core';
import {Router} from '@angular/router';
import {NzModalService} from 'ng-zorro-antd/modal';
import {
  CaseStudiesModalCreateComponent
} from 'src/app/components/data/case-studies/case-studies-modal-create/case-studies-modal-create.component';
import {DynamicBrowseComponent} from 'src/app/components/dynamics/dynamic-browse/dynamic-browse.component';
import {AuthService} from 'src/app/services/auth.service';
import {BackendService} from 'src/app/services/backend.service';
import {MessageLogService} from 'src/app/services/message-log.service';
import {StateService} from 'src/app/services/state.service';
import {AclModalComponent} from 'src/app/components/admin/acl/acl-modal/acl-modal.component';
import {
  GeolayerExportModalComponent
} from 'src/app/components/data/geolayer/geolayer-export-modal/geolayer-export-modal.component';
import {TopAreaService} from '../../../services/top-area.service';
import {NotificationService, TypeNotificationEnum} from 'src/app/services/notification.service';
import {
  CaseStudiesAddItemsModalComponent
} from "src/app/components/data/case-studies/case-studies-add-items-modal/case-studies-add-items-modal.component";
import {
  CollectionsAddItemsModalComponent
} from "src/app/components/data/collection/collections-add-items-modal/collections-add-items-modal.component";
import {ModalInfoService} from "../../../services/modal-info.service";
import {CommonModule} from '@angular/common';
import {InfoModalComponent} from "../../../components/miscellaneous/info-modal/info-modal.component";
import {
  GenerateLayerViewerModalComponent
} from "../../../components/data/geolayer/generate-layer-viewer-modal/generate-layer-viewer-modal.component";
import IView from '@interfaces/view.interfaces';
import {lastValueFrom} from "rxjs";

@Component({
    selector: 'app-geolayer-browse',
    imports: [
        CommonModule,
        CaseStudiesAddItemsModalComponent,
        CollectionsAddItemsModalComponent,
        InfoModalComponent,
        GenerateLayerViewerModalComponent,
        CaseStudiesModalCreateComponent,
        GeolayerExportModalComponent,
        AclModalComponent,
        DynamicBrowseComponent,
    ],
    templateUrl: './geolayer-browse.component.html',
    styleUrls: ['./geolayer-browse.component.sass']
})
export class GeolayerBrowseComponent implements OnInit, AfterViewInit, OnDestroy, IView {

  @ViewChild(DynamicBrowseComponent)
  private dynamicBrowser: DynamicBrowseComponent;

  BREADCRUMB_NAME = "GEOGRAPHICAL_DATA.LAYERS.BROWSER.BREADCRUMB";

  loading = false;
  customFilterRequest: any = {};

  // CUSTOM BUTTONS BROWSER
  customButtonsBrowser = [];
  @ViewChild(CaseStudiesAddItemsModalComponent) caseStudiesAddItemsModalComponent: CaseStudiesAddItemsModalComponent;
  @ViewChild(CollectionsAddItemsModalComponent) collectionsAddItemsModalComponent: CollectionsAddItemsModalComponent;

  // ACL
  @ViewChild('modalAclRef', {static: false}) modalAclRef: AclModalComponent;
  isVisibleAclModal = false;

  // Modal Export Layers
  @ViewChild(GeolayerExportModalComponent) geolayerExportModal: GeolayerExportModalComponent;
  @ViewChild(CaseStudiesModalCreateComponent) caseStudiesCreateModal: CaseStudiesModalCreateComponent;

  // Generate Layer
  @ViewChild(GenerateLayerViewerModalComponent) generateLayerViewerModalComponent: GenerateLayerViewerModalComponent;

  // Generate view Modal
  isLoadingGenerateViewModal = false;
  isVisibleGenerateViewModal = false;
  isVisibleGenerateViewErrorModal;
  layersGenerateViewModal = [];

  constructor(
    private readonly router: Router,
    private readonly stateService: StateService,
    private readonly backendService: BackendService,
    private readonly logService: MessageLogService,
    private modalService: NzModalService,
    public readonly authService: AuthService,
    private readonly topAreaService: TopAreaService,
    private readonly notificationService: NotificationService,
    private readonly modalInfoService: ModalInfoService
  ) {
  }

  ngOnInit(): void {
    this.createCustomButtonsBrowser();
  }

  async ngAfterViewInit(): Promise<void> {
    this.loading = true;
    const getState = await this.stateService.getStateCurrenView();
    if (getState) {
      const dynamicTable = this.dynamicBrowser.dynamicTable;
      if (getState.pageIndex) {
        dynamicTable.pageIndex = getState.pageIndex;
      }
      if (getState.pageSize) {
        dynamicTable.pageSize = getState.pageSize;
      }
      this.checkItemsByState(getState);
      this.dynamicBrowser.changePage();
    }
    this.loading = false;
  }

  async checkItemsByState(state): Promise<void> {
    const dynamicTable = this.dynamicBrowser.dynamicTable;
    const response: any = await lastValueFrom(this.backendService.getLayerGIS(null, {only_ids: true}), { defaultValue: undefined });
    const content = response.content as any[];
    if (state.checkedItems && state.checkedItems instanceof Array && state.checkedItems.length > 0) {
      const intersection = state.checkedItems.filter(item => {
        return content.includes(item);
      });
      dynamicTable.checkedItems = new Set(intersection);
    }
  }

  ngOnDestroy(): void {
    for (const button of this.customButtonsBrowser) {
      button.eventEmitter.unsubscribe();
    }
    const dynamicTable = this.dynamicBrowser.dynamicTable;
    this.stateService.setStateCurrentView({
      pageIndex: dynamicTable.pageIndex,
      pageSize: dynamicTable.pageSize,
      checkedItems: Array.from(dynamicTable.checkedItems),
    });
  }

  private createCustomButtonsBrowser(): void {
    // Create
    const addItemCasesStudies = {
      title: 'STUDY_CASES.ADD_ITEM_CUSTOM_BUTTON_BROWSERS.TEXT',
      needCheckedItem: true,
      eventEmitter: new EventEmitter(),
      cantCheckedAllItem: true,
      funcCondition: () => {
        return this.authService.havePermission('gui-case-study-edit');
      }
    };

    const addItemCollection = {
      title: 'MOLECULAR_DATA.COLLECTIONS.ADD_BUTTON_ITEM',
      needCheckedItem: true,
      eventEmitter: new EventEmitter(),
      cantCheckedAllItem: true,
      funcCondition: () => {
        return this.authService.havePermission('gui-collection-edit');
      }
    }

    const exportLayers = {
      title: 'GEOGRAPHICAL_DATA.LAYERS.BROWSER.CUSTOM_BUTTONS.EXPORT_LAYERS.TITLE',
      needCheckedItem: true,
      eventEmitter: new EventEmitter(),
      cantCheckedAllItem: true,
      funcCondition: () => {
        return this.authService.havePermission('gui-layer-export');
      }
    };

    const createLayer = {
      title: 'GEOGRAPHICAL_DATA.LAYERS.BROWSER.CUSTOM_BUTTONS.CREATE_LAYER.TITLE',
      needCheckedItem: true,
      eventEmitter: new EventEmitter(),
      cantCheckedAllItem: true,
      funcCondition: () => {
        return this.authService.havePermission('gui-layer-import');
      }
    }

    const openInGis = {
      title: 'GEOGRAPHICAL_DATA.LAYERS.BROWSER.CUSTOM_BUTTONS.OPEN_IN_GIS.TITLE',
      needCheckedItem: true,
      eventEmitter: new EventEmitter(),
      cantCheckedAllItem: true,
      funcCondition: () => {
        return this.authService.havePermission('gui-layers-graphical');
      }
    }

    // Events
    addItemCasesStudies.eventEmitter.subscribe({next:async (event) => {
      this.loading = true;
      try {
        const response: any = await this.getLayersByChecks(event);
        const entitiesId = response.content.map((entity) => {
          return entity.id;
        });
        await this.caseStudiesAddItemsModalComponent.openModal(entitiesId);
      } catch (e) {
        this.modalInfoService.showModalInfoDefaultError(e);
      }
      this.loading = false;
    }});

    addItemCollection.eventEmitter.subscribe({next:async (event) => {
      this.loading = true;
      try {
        const response: any = await this.getLayersByChecks(event);
        const entitiesUuid = response.content.map((entity) => {
          return entity.uuid;
        });
        await this.collectionsAddItemsModalComponent.openModal(entitiesUuid);
      } catch (e) {
        this.modalInfoService.showModalInfoDefaultError(e);
      }
      this.loading = false;
    }})

    exportLayers.eventEmitter.subscribe({next:async (event) => {
      this.loading = true;
      const layersResponse = await lastValueFrom(this.backendService.getLayerGIS(null, {
        filter: event,
      }), { defaultValue: undefined });
      const layersContent = layersResponse["content"];
      await this.geolayerExportModal.openModal(layersContent);
      this.loading = false;
    }});

    createLayer.eventEmitter.subscribe({next:async (event) => {
      this.loading = true;
      await this.openGenerateViewModal(event);
      this.loading = false;
    }});

    openInGis.eventEmitter.subscribe({next:async (event) => {
      this.loading = true;
      try {
        const response: any = await lastValueFrom(this.backendService.getLayerGIS(null, {filter: event}));
        const responseContent: any [] = response.content;
        if (responseContent.length > 0) {
          let uuidLayersQueryParam = "";
          responseContent.forEach((element, index) => {
            if (index !== 0) {
              uuidLayersQueryParam += ',';
            }
            uuidLayersQueryParam += `${element.uuid}`;
          });
          this.router.navigate(['phylogeographyBrowse'], {
            queryParams: {
              uuidLayers: uuidLayersQueryParam,
            }
          }).then();
        }
      } catch (e) {
        this.modalInfoService.showModalInfoDefaultError(e);
      }
      this.loading = false;
    }});

    // Add list
    this.customButtonsBrowser = [
      addItemCasesStudies,
      addItemCollection,
      exportLayers,
      createLayer,
      openInGis,
    ];
  }

  async getLayersByChecks(event): Promise<any> {
    const response: any = await lastValueFrom(this.backendService.getLayerGIS(null, {
      filter: event,
    }), { defaultValue: undefined });
    if (!response.content || response.content.length === 0) {
      this.loading = false;
      throw {message: "Error GET SEQUENCES"};
    }
    return response;
  }

  createCaseStudySuccessEvent() {
    this.dynamicBrowser.dynamicTable.checkedItems.clear();
  }

  importLayer() {
    this.router.navigate(['geolayerImport']);
  }

  hiddenImport(): boolean {
    return !this.authService.havePermission('gui-layer-import');
  }

  async openLayer(data): Promise<void> {
    this.router.navigate(['geolayerDetail', data.id]);
  }

  async removeItemLayer(data): Promise<void> {
    this.modalService.confirm({
      nzTitle: '<i><i>¿Seguro que desea eliminar este elemento?</i></i>',
      nzContent: `<b>Capa: ${data.name}</b>`,
      nzCentered: true,
      nzOnOk: () => {
        this.loading = true;
        this.backendService.deleteLayerGIS(data.id).subscribe({next:(res) => {
          this.logService.addIssues(res['issues']);
        }, error: (err) => {
          this.loading = false;
          this.logService.addIssues(err['issues']);
        }, complete: () => {
          this.loading = false;
          this.dynamicBrowser.changePage();
        }});
      }
    });
  }

  //PERMISSION
  permissionItem(item) {
    this.modalAclRef.featureId = item.id;
    this.modalAclRef.objectTypeName = "geolayer";
    this.isVisibleAclModal = true;
    this.modalAclRef.objectUUID = item.uuid;
  }

  async openGenerateViewModal(filter?): Promise<void> {
    this.loading = true;
    const response: any = await lastValueFrom(this.backendService.getLayerGIS(null, {filter: filter}), { defaultValue: undefined });
    this.generateLayerViewerModalComponent.Layers = response.content;
    if (this.generateLayerViewerModalComponent.Layers.length === 0) {
      this.notificationService.createNotificationWithType(
        TypeNotificationEnum.warning,
        'DISPLAY.GIS.LAYERS_EMPTY_OPEN_GENERATE_VIEW_MODAL_ERROR.TITLE',
        'DISPLAY.GIS.LAYERS_EMPTY_OPEN_GENERATE_VIEW_MODAL_ERROR.CONTENT',
        'bottomRight'
      );
      this.loading = false;
      return;
    }
    this.generateLayerViewerModalComponent.isVisible = true;
    this.loading = false;
  }

  async onClickAcceptGenerateLayer(event): Promise<void> {
    this.isLoadingGenerateViewModal = true;
    try {
      console.log(event);
      const response: any = await lastValueFrom(this.backendService.postLayerGISByQuery(event), { defaultValue: undefined });
      if (response.issues) {
        this.logService.addIssues(response.issues);
      }
      this.isVisibleGenerateViewModal = false;
    } catch (e) {
      if (e) {
        this.isVisibleGenerateViewErrorModal = true;
        if (e.issues) {
          this.logService.addIssues(e.issues);
        }
      }
    }
    this.isLoadingGenerateViewModal = false;
    this.generateLayerViewerModalComponent.isVisible = false
    this.dynamicBrowser.changePage();
  }

}
