import { Component, Input, OnInit } from '@angular/core';
import { ActivatedRoute, NavigationEnd, Router, RouterModule } from '@angular/router';
import { AuthService } from '@services/auth.service';
import { SidebarService } from '@services/sidebar.service';
import { CommonModule } from '@angular/common';
import { SharedModule } from "@shared-module/shared.module";
import { NzMenuModule } from "ng-zorro-antd/menu";
import { KeyValueCustomPipe } from "@pipes/key-value-custom.pipe";
import { SidebarItem } from '@interfaces/sidebar-item.interfaces';
import { FormsModule } from '@angular/forms';
import { NzInputModule } from 'ng-zorro-antd/input';
import { TranslateService } from '@ngx-translate/core';
import { I } from '@angular/cdk/a11y-module.d-DBHGyKoh';
/**
 * @component SidebarComponent
 * @description
 * Componente responsable de renderizar el menú lateral (sidebar) de la aplicación.
 * Carga dinámicamente las secciones y rutas desde un archivo YAML mediante el `SidebarService`.
 * También gestiona la apertura/cierre de submenús y el resaltado del elemento activo según la ruta actual.
 */
@Component({
  selector: 'app-sidebar',
  imports: [
    CommonModule,
    SharedModule,
    NzMenuModule,
    RouterModule,
    KeyValueCustomPipe,
    FormsModule,
    NzInputModule
  ],
  templateUrl: './sidebar.component.html',
  styleUrls: ['./sidebar.component.sass']
})

export class SidebarComponent implements OnInit {
  
  @Input() isCollapsable: boolean = false;
  /** Estructura completa del sidebar cargada desde el YAML */
  items: SidebarItem | null;

  show : boolean = true;

  filteredItems: any = {};

  /** Ruta actual del navegador */
  currentUrl = '';

  /** Valor especial para rutas personalizadas de procesos o geoprocesos */
  customRawUrlSelect = null;

  /** Mapa que controla qué submenús están abiertos */
  openMap: { [name: string]: boolean } = {};

  /** Texto de búsqueda para filtrar elementos del sidebar */
  searchText: string = '';
  lastSearchText: string = '';

  lastState: { [name: string]: boolean } = {};
  /**
   * Cierra todos los submenús excepto el seleccionado.
   * @param value Nombre del submenú que se desea mantener abierto
   */
  openHandler(value: string): void {
    for (const key in this.openMap) {
      if (key !== value) {
        this.openMap[key] = false;
        this.lastState = { ...this.openMap };
      }
    }
  }

  /**
   * Constructor del componente Sidebar
   * @param activeRoute Manejador de la ruta activa actual
   * @param router Servicio Angular para la navegación
   * @param sidebarService Servicio para cargar la configuración del sidebar desde YAML
   * @param authService Servicio de autenticación y permisos
   */
  constructor(
    public activeRoute: ActivatedRoute,
    private router: Router,
    private sidebarService: SidebarService,
    public authService: AuthService,
    private translate: TranslateService,
  ) { }

  /**
   * Inicializa el componente cargando el sidebar desde el archivo YAML,
   * escuchando los eventos de navegación y los eventos personalizados del sidebar.
   */
  async ngOnInit(): Promise<void> {
    this.items = await this.sidebarService.loadSidebarItems('/assets/config/subDemo.yaml');
    this.filteredItems = { ...this.items };


    if (this.items) {
      Object.keys(this.items).forEach(submenu => {
        this.openMap[submenu] = false;
      });
    }
    this.lastState = { ...this.openMap };

    // Escucha eventos personalizados emitidos por SidebarService
    this.sidebarService.sidebarEventEmitter.subscribe((event: { type: string, value: any }) => {
      switch (event.type) {
        case 'customRawUrlSelect':
          this.customRawUrlSelect = event.value;
          if (this.customRawUrlSelect) {
            this.openMenuDependCurrentView();
          }
          break;
      }
    });

    // Escucha los cambios de ruta y actualiza el menú activo
    this.router.events.subscribe((event) => {
      if (event instanceof NavigationEnd) {
        this.currentUrl = this.router.url;
        this.openMenuDependCurrentView();
      }
    });
  }

  /**
   * Determina si un submenú debe mostrarse según los permisos del usuario.
   * @param subMenu Submenú a evaluar
   * @returns `true` si el usuario tiene permiso o el submenú no requiere permisos
   */
  showSubMenu(subMenu: SidebarItem): boolean {
    const routes = subMenu.routes;
    for (const key in routes) {
      const route = routes[key];
      if (!route.permission) {
        return true;
      }
      if (this.funcCondition(route.permission)) {
        return true;
      }
    }
    return false;
  }

  /**
   * Abre el submenú correspondiente según la ruta actual del navegador.
   * Este método mantiene el menú sincronizado con la vista activa.
   * @private
   */
  private openMenuDependCurrentView() {
    const openMenu = (value: string) => {
      this.openMap[value] = true;
      this.openHandler(value);
    };

    for (const submenu in this.items) {
      for (const keyItem in this.items[submenu].routes) {
        for (const route of this.items[submenu].routes[keyItem].routes) {
          if (this.customRawUrlSelect &&
            (this.currentUrl.startsWith('/process/processDetail') ||
              this.currentUrl.startsWith('/geoprocessInstancesDetail'))) {
            if (this.customRawUrlSelect.startsWith(route)) {
              openMenu(submenu);
              return;
            }
          } else if (this.currentUrl.startsWith(route)) {
            openMenu(submenu);
            return;
          }
        }
      }
    }
  }

  /**
   * Determina si un ítem del menú está actualmente seleccionado.
   * @param item Ruta o ítem del menú a evaluar
   * @returns `true` si coincide con la ruta actual
   */
  isSelectedItem(item): boolean {
    if (this.currentUrl.startsWith('/process/processDetail') ||
      this.currentUrl.startsWith('/geoprocessInstancesDetail')) {
      if (this.customRawUrlSelect) {
        for (const route of item.routes) {
          if (this.customRawUrlSelect.startsWith(route)) { return true; }
        }
      }
      return false;
    }
    for (const route of item.routes) {
      if (this.currentUrl.startsWith(route)) { return true; }
    }
    return false;
  }

  /**
   * Obtiene el texto o código de traducción del ítem.
   * @param item Elemento del sidebar
   * @returns Código de traducción o clave por defecto
   */
  getText(item: SidebarItem | { key: string; value: any }): string {
    return item.value?.code ? item.value.code : `SIDEBAR.${item.key}.title`;
  }



  /**
   * Evalúa si el usuario posee un permiso específico.
   * @param perm Clave del permiso a evaluar
   * @returns `true` si el usuario tiene el permiso
   */
  funcCondition(perm: string): boolean {
    return this.authService.havePermission(perm);
  }

  search(query: string): void {
    console.log('Buscar en el sidebar:', query);
  }

  onSearchChange() {
    if (this.searchText === '') {
      this.openMap = { ...this.lastState };
    }
    const query = this.searchText.trim().toLowerCase();

    if (!query) {
      this.filteredItems = { ...this.items };
      return;
    }

    const result: any = {};

    for (const [subKey, subMenu] of Object.entries(this.items)) {
      // Buscar coincidencias en las rutas internas
      const filteredRoutes = Object.entries(subMenu.routes || {}).filter(
        ([routeKey, routeValue]: [string, any]) => {
          const itemWrapper = { key: routeKey, value: routeValue };
          const code = this.getText(itemWrapper);
          const translated = this.translate.instant(code)?.toLowerCase() || '';
          return (
            code.toLowerCase().includes(query) ||
            translated.includes(query) ||
            routeKey.toLowerCase().includes(query)
          );
        }
      );

      // Buscar coincidencia en el título o nombre del submenú
      const subMenuCode = subMenu.code || `SIDEBAR.${subKey}.title`;
      const subMenuTranslated = this.translate.instant(subMenuCode)?.toLowerCase() || '';
      const subMenuMatches =
        subMenuCode.toLowerCase().includes(query) ||
        subMenuTranslated.includes(query) ||
        subKey.toLowerCase().includes(query);

      // Si el submenú coincide, mostrar todas sus rutas; si no, solo las filtradas
      if (subMenuMatches || filteredRoutes.length > 0) {
        result[subKey] = {
          ...subMenu,
          routes: subMenuMatches
            ? subMenu.routes // mostrar todas las rutas si el título del submenú coincide
            : Object.fromEntries(filteredRoutes)
        };
      }
    }

    this.filteredItems = result;
    for (const item in result) {
      this.openMap[item] = true;
    }
  }
  clearSearch() {
    this.searchText = '';
    this.onSearchChange(); // opcional, por si quieres refrescar resultados
  }
}
