import {
  AfterViewChecked,
  ChangeDetectorRef,
  Component,
  ElementRef,
  EventEmitter,
  OnDestroy,
  OnInit,
  ViewChild
} from '@angular/core';
import {ActivatedRoute, Router} from '@angular/router';
import {NzModalRef, NzModalService} from 'ng-zorro-antd/modal';
import {Subject} from 'rxjs';
import {DynamicBrowseComponent} from 'src/app/components/dynamics/dynamic-browse/dynamic-browse.component';
import {AuthService} from 'src/app/services/auth.service';
import {BackendService} from 'src/app/services/backend.service';
import {MessageLogService} from 'src/app/services/message-log.service';
import {StateService} from '@services/state.service';
import {
  OrganizationFormComponent
} from 'src/app/components/admin/organizations/organization-form/organization-form.component';
import IView from "../../../../interfaces/view.interfaces";
import {CommonModule} from '@angular/common';
import {NzButtonModule} from "ng-zorro-antd/button";
import {SharedModule} from "../../../../shared-module/shared.module";

@Component({
    selector: 'app-organizations-browser',
    imports: [
        CommonModule,
        NzButtonModule,
        OrganizationFormComponent,
        DynamicBrowseComponent,
        SharedModule,
    ],
    templateUrl: './organizations-browser.component.html',
    styleUrls: ['./organizations-browser.component.sass']
})
export class OrganizationsBrowserComponent implements OnInit, OnDestroy, AfterViewChecked, IView {

  BREADCRUMB_NAME = 'SECURITY_ADMIN.ORGANIZATIONS.BROWSER.BREADCRUMB';
  state;
  buttons =[];

  @ViewChild("containerPage", {static: false}) containerPage: ElementRef;
  @ViewChild(DynamicBrowseComponent) dynamicBrowser: DynamicBrowseComponent;

  page = 1;
  pageSize = 10;
  loading = false;
  pageData: any = [];
  totalItems = 0;
  searchValue: string = "";
  excludeFields = ['identities'];

  // Create Modal
  @ViewChild("modalCreateTitle", {static: false}) modalCreateTitle;
  @ViewChild("modalCreateContent", {static: false}) modalCreateContent;
  @ViewChild("organizationFormElementRef", {static: false}) organizationFormElementRef: OrganizationFormComponent;
  @ViewChild("modalCreateFooter", {static: false}) modalCreateFooter;
  disableEditFormSubject: Subject<boolean> = new Subject<boolean>();

  constructor(
    private readonly backendService: BackendService,
    private readonly msg: MessageLogService,
    private readonly stateService: StateService,
    private readonly router: Router,
    private readonly modalService: NzModalService,
    private readonly activeRoute: ActivatedRoute,
    private readonly cdRef: ChangeDetectorRef,
    public readonly authService: AuthService,
  ) {
  }

  ngOnInit(): void {
    this.createCustomButtonsBrowser();
    const ORGANIZATIONS_DATA = this.activeRoute.snapshot.data.organizations;
    this.pageData = ORGANIZATIONS_DATA["content"];
    this.totalItems = ORGANIZATIONS_DATA["count"];
  }

  async ngAfterViewInit(): Promise<void> {
    this.loading = true;
    const getState = await this.stateService.getStateCurrenView();
    if (getState) {
      const dynamicTable = this.dynamicBrowser.dynamicTable
      if (getState.pageIndex) {
        dynamicTable.pageIndex = getState.pageIndex;
      }
      if (getState.pageSize) {
        dynamicTable.pageSize = getState.pageSize;
      }
      if (getState.checkedItems && getState.checkedItems instanceof Array) {
        dynamicTable.checkedItems = new Set(getState.checkedItems);
      }
      this.dynamicBrowser.changePage();
    }
    this.loading = false;
  }

  ngOnDestroy(): void {
    for (const button of this.buttons) {
      button.eventEmitter.unsubscribe();
    }
    const dynamicTable = this.dynamicBrowser.dynamicTable;
    this.stateService.setStateCurrentView({
      pageIndex: dynamicTable.pageIndex,
      pageSize: dynamicTable.pageSize,
      checkedItems: Array.from(dynamicTable.checkedItems),
    });
  }

  private createCustomButtonsBrowser(): void {

  }

  ngAfterViewChecked() {
    this.cdRef.detectChanges();
  }

  private search() {
    this.loading = true;
    this.dynamicBrowser.changePage();
    this.loading = false;
  }

  onClickCreateEvent(): void {
    this.loading = true;
    this.openCreateOrganizationModal();
    this.loading = false;
  }

  calculeScrollTableY() {
    if (this.containerPage) {
      const containerDivPage = <HTMLDivElement>this.containerPage.nativeElement;
      var scrollTableY = containerDivPage.offsetHeight - 350;
      if (scrollTableY < 0) scrollTableY = 0;
      return `${scrollTableY}px`;
    }
    return '0px';
  }

  //   if (checked) {
  //     this.setOfCheckedId.add(id);
  //   } else {
  //     this.setOfCheckedId.delete(id);
  //   }
  // }

  // onItemChecked(id: number, checked: boolean): void {
  //   this.updateCheckedSet(id, checked);
  // }

  searchEvent() {
    this.page = 1;
    this.search();
  }

  changePage(page) {
    this.page = page;
    this.search()
  }

  changePaginationEvent(event) {
    this.page = event.pageIndex;
    this.search();
  }

  openItem(item) {
    this.router.navigate(['organizationDetail', item.id]);
  }

  modifyItem(item) {
    this.router.navigate(['organizationModify', item.id]);
  }

  deleteItem(item) {
    this.modalService.confirm({
      nzTitle: '¿Seguro que desea eliminar este grupo?',
      nzContent: item.name,
      nzOkText: 'Eliminar',
      nzOkType: 'primary',
      nzOkDanger: true,
      nzCentered: true,
      nzOnOk: () => this.deleteOrganization(item.id),
      nzCancelText: 'Cancelar',
    });
  }

  deleteOrganization(id) {
    this.backendService.deleteOrganization(id).subscribe({next: (response) => {
      console.log(response);
      this.search();
    }, error: (error) => {
      console.log(error);
    }});
  }

  openCreateOrganizationModal() {
    const modal = this.modalService.create({
      nzTitle: this.modalCreateTitle,
      nzContent: this.modalCreateContent,
      nzFooter: this.modalCreateFooter,
      nzMaskClosable: false,
    });
    modal.afterOpen.subscribe({next:() => {
      this.disableEditFormSubject.next(false);
    }});
  }

  onClickCreateOrganizationModal() {
    this.organizationFormElementRef.createOrganization();
  }

  disableCreateButtonOrganizationModal() {
    if (!this.organizationFormElementRef) {
      return true;
    } else {
      return !this.organizationFormElementRef.formGroup.valid;
    }
  }

  requestEndCreateOrganization(event: {
    methodRequest: 'PUT' | 'POST',
    responseType: 'error' | 'success',
    data?: any
  }) {
    if (event.methodRequest == "POST") {
      if (event.responseType == 'success') {
        this.modalService.closeAll();
        this.search();
      }
    }
  }

  onClickDestroyModal(modalRef: NzModalRef) {
    modalRef.destroy();
  }

}
