import {Component, OnDestroy, OnInit, ViewChild} from '@angular/core';
import {FormsModule, ReactiveFormsModule, UntypedFormBuilder, UntypedFormGroup} from '@angular/forms';
import {ActivatedRoute, Router} from '@angular/router';
import IView from '@interfaces/view.interfaces';
import {AuthService} from 'src/app/services/auth.service';
import {BackendService} from 'src/app/services/backend.service';
import {StateService} from 'src/app/services/state.service';
import {MessageLogService} from '../../../../services/message-log.service';
import {CommonModule} from '@angular/common';
import {
  AnnotationsFormComponent
} from "../../../../components/data/annotations/annotations-form/annotations-form.component";
import {SharedModule} from "../../../../shared-module/shared.module";
import {NzTabsModule} from "ng-zorro-antd/tabs";
import {BOCardComponent} from "../../../../components/miscellaneous/bo-card/bo-card.component";
import {NzTableModule} from "ng-zorro-antd/table";
import {DisableControlDirective} from "../../../../directives/disable-control.directive";
import {NzFormModule} from "ng-zorro-antd/form";
import {NzInputModule} from "ng-zorro-antd/input";
import {NzButtonModule} from "ng-zorro-antd/button";
import {NzSelectModule} from "ng-zorro-antd/select";
import {NzSpinModule} from "ng-zorro-antd/spin";
import {BackButtonComponent} from "../../../../components/miscellaneous/back-button/back-button.component";
import {NzPageHeaderModule} from "ng-zorro-antd/page-header";
import {
  SelectLazyLoadingComponent
} from 'src/app/components/miscellaneous/select-lazy-loading/select-lazy-loading.component';
import {forkJoin, lastValueFrom} from "rxjs";

@Component({
    selector: 'app-individual-detail',
    imports: [
        CommonModule,
        AnnotationsFormComponent,
        SharedModule,
        NzTabsModule,
        BOCardComponent,
        NzTableModule,
        DisableControlDirective,
        NzFormModule,

        NzPageHeaderModule,
        NzInputModule,
        SelectLazyLoadingComponent,
    ],

    styleUrls: ['./individual-detail.component.sass']
})
export class IndividualDetailComponent implements OnInit, OnDestroy, IView {

  BREADCRUMB_NAME = {
    key: 'METADATA.INDIVIDUALS.DETAIL.BREADCRUMB',
    params: {
      name: '',
    }
  };

  individual;
  generalFormGroup: UntypedFormGroup;
  generalFormGroupChange = false;
  loadingGeneral = false;
  organisms: any[] = [];
  cvterm: any = {};
  editing = false;
  sequences: any[] = [];
  analyses: any[] = [];
  processes: any = [];

  @ViewChild('annotationsForm', {static: false}) annotationsForm: AnnotationsFormComponent;

  constructor(
    private readonly activatedRoute: ActivatedRoute,
    private readonly stateService: StateService,
    private readonly fb: UntypedFormBuilder,
    public readonly authService: AuthService,
    private readonly backendService: BackendService,
    private readonly router: Router,
    private readonly messageLogServices: MessageLogService,
  ) {

  }

  async ngOnInit(): Promise<void> {
    this.loadingGeneral = true;
    this.individual = this.activatedRoute.snapshot.data.individual;
    this.initGeneralFormGroup();
    const getState = await this.stateService.getStateCurrenView();
    await this.resetGeneralFormGroup();
    if (getState && getState.generalFormGroupChange) {
      this.generalFormGroup.get('name').setValue(getState.generalFormGroupName ?
        getState.generalFormGroupName : this.individual.name);
      this.generalFormGroup.get('uniquename').setValue(getState.generalFormGroupUniqueName ?
        getState.generalFormGroupUniqueName : this.individual.uniquename);
      this.generalFormGroupChange = getState.generalFormGroupChange;
    }
    this.loadingGeneral = false;
  }

  ngOnDestroy(): void {
    if (!this.generalFormGroupChange) {
      this.stateService.setStateCurrentView({
        generalFormGroupChange: this.generalFormGroupChange,
      }).then();
    } else {
      this.stateService.setStateCurrentView({
        name: this.generalFormGroup.get('name').value ? this.generalFormGroup.get('name').value : '',
        uniquename: this.generalFormGroup.get('uniquename').value ? this.generalFormGroup.get('uniquename').value : '',
        generalFormGroupChange: this.generalFormGroupChange,
      }).then();
    }
  }

  initGeneralFormGroup() {
    this.generalFormGroup = this.fb.group({
      stock_id: [{value: this.individual.stock_id, disabled: true}],
      name: [''],
      uniquename: [''],
      organism_id: [this.individual.organism_id],
      description: [''],
      cvterm_id: [{value: this.individual.cvterm_id, disabled: true}],
    });
    this.generalFormGroup.valueChanges.subscribe({next: () => {
      this.generalFormGroupChange = true;
    }});
  }

  ngAfterViewInit(): void {
    this.annotationsForm.object_uuid = this.individual.uuid;
    this.annotationsForm.loadAnnotations();
  }

  editEvent() {
    this.editing = !this.editing;
  }

  isDisableEdit(): boolean {
    return !this.authService.havePermission('gui-individual-edit');
  }

  async modifyItem(): Promise<void> {
    this.loadingGeneral = true;
    try {
      const value = {
        uniquename: this.generalFormGroup.get('uniquename').value,
        name: this.generalFormGroup.get('name').value,
        organism_id: this.generalFormGroup.get('organism_id').value,
        description: this.generalFormGroup.get('description').value,
        cvterm_id: this.generalFormGroup.get('cvterm_id').value,
      };
      const response: any = await lastValueFrom(this.backendService.putIndividual(this.individual.stock_id, value), { defaultValue: undefined });
      this.messageLogServices.addIssues(response.issues);
      await this.resetGeneralFormGroup();
    } catch (e) {
      this.messageLogServices.addIssues(e.issues);
    }
    this.loadingGeneral = false;
  }

  async cancelModifyItem(): Promise<void> {
    this.loadingGeneral = true;
    try {
      await this.resetGeneralFormGroup();
    } catch (e) {
      if (e.issues) {
        this.messageLogServices.addIssues(e.issues);
      }
    }
    this.loadingGeneral = false;
    this.generalFormGroupChange = false;
  }

  async resetGeneralFormGroup(): Promise<void> {
    const response: any = await lastValueFrom(forkJoin({
      cvterm: this.backendService.getCvterms(undefined, this.individual.type_id),
      sequences: this.backendService.getSequences(undefined,
        this.backendService.filter({stock_id: this.individual.stock_id})),
      analyses: this.backendService.getAnalyses(undefined,
        this.backendService.filter({stock_id: this.individual.stock_id})),
      individual: this.backendService.getIndividuals(this.individual.stock_id),
    }));
    this.cvterm = response.cvterm.content;
    this.messageLogServices.addIssues(response.cvterm.issues);
    this.sequences = response.sequences.content;
    this.messageLogServices.addIssues(response.sequences.issues);
    this.analyses = response.analyses.content;
    this.messageLogServices.addIssues(response.analyses.issues);
    const individual = response.individual.content;
    this.messageLogServices.addIssues(response.individual.issues);

    this.individual.name = individual.name ? individual.name : '';
    this.individual.uniquename = individual.uniquename ? individual.uniquename : '';
    this.individual.organism_id = individual.organism_id ? individual.organism_id : null;
    this.individual.description = individual.description ? individual.description : '';
    this.individual.cvterm_id = individual.cvterm_id ? individual.cvterm_id : null;
    this.generalFormGroup.get('name').setValue(this.individual.name);
    this.generalFormGroup.get('uniquename').setValue(this.individual.uniquename);
    this.generalFormGroup.get('organism_id').setValue(this.individual.organism_id);
    this.generalFormGroup.get('description').setValue(this.individual.description);
    this.generalFormGroup.get('cvterm_id').setValue(this.individual.cvterm_id);
    this.BREADCRUMB_NAME.params.name = this.generalFormGroup.get('uniquename').value;
    this.generalFormGroupChange = false;
  }

  organismValueChange(event): void {
    console.log(event);
    this.generalFormGroup.get('organism_id').setValue(event);
  }

  getMessageUniquenameError(): string[] | null {
    const message = [];
    if (!this.generalFormGroup.errors || !this.generalFormGroup.errors.uniquenameError) {
      return null;
    }
    if (this.generalFormGroup.errors.uniquenameError.empty) {
      message.push('METADATA.INDIVIDUALS.DETAIL.UNIQUENAME_INPUT.ERROR_EMPTY');
    }
    return message;
  }

  goOrganismDetail(): void {
    this.router.navigate(['organismDetail', this.generalFormGroup.get('organism_id').value]);
  }

  goOntologyDetail(): void {
    this.router.navigate(['termDetail', this.generalFormGroup.get('cvterm_id').value]);
  }

  navigateSequence(id) {
    if (!this.authService.havePermission('gui-seq-read')) {
      return;
    }
    this.router.navigate(['sequenceDetail', id]);
  }

  goAnalysisDetail(id) {
    if (!this.authService.havePermission('gui-analysis-read')) {
      return;
    }
    this.router.navigate(['analysisDetail', id]);
  }

}
