import { Component } from '@angular/core';
import { BackendService } from 'src/app/services/backend.service';
import {UntypedFormGroup, UntypedFormControl, FormsModule, ReactiveFormsModule} from '@angular/forms';
import {MessageLogService} from "../../../../services/message-log.service";
import { CommonModule } from '@angular/common';
import {SharedModule} from "../../../../shared-module/shared.module";
import {NzUploadModule} from "ng-zorro-antd/upload";
import {NzButtonModule} from "ng-zorro-antd/button";
import {NzFormModule} from "ng-zorro-antd/form";
import {NzPageHeaderModule} from "ng-zorro-antd/page-header";

@Component({
    selector: 'app-taxonomies-import',
    imports: [
        CommonModule,
        SharedModule,
        NzUploadModule,
        NzButtonModule,
        NzFormModule,
        FormsModule,
        ReactiveFormsModule,
        NzPageHeaderModule,
    ],
    templateUrl: './taxonomies-import.component.html',
    styleUrls: ['./taxonomies-import.component.sass']
})
export class TaxonomiesImportComponent {

  constructor(public backendService: BackendService, private logService: MessageLogService) {}

  fileList = [];
  dataForm: UntypedFormGroup = new UntypedFormGroup({
    name: new UntypedFormControl(),
    comment: new UntypedFormControl({value: '', disabled: true}),
    taxa: new UntypedFormControl()
  });
  loading = false;

  beforeUpload = (file): boolean => {
    this.fileList = this.fileList.concat(file);
    return false;
  }

  removeFile = (event): void => {
    for (let index = 0; index < this.fileList.length; index++) {
      if (event['uid'] == this.fileList[index]['uid']) {
        this.fileList.splice(index, 1);
      }
    }
  }

  importFiles(): void {
    this.loading = true;
    this.backendService.importTaxonomies(this.fileList, this.dataForm.value).subscribe(
      {next: response => {
        this.logService.addIssues(response['issues']);
      }, error: error => {
        this.loading = false;
        this.logService.addIssues(error['issues']);
      }, complete: () => {
        this.loading = false;
        this.fileList = [];
      }});
  }
}
