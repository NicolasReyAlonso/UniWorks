import {Component, inject, ViewChild} from '@angular/core';
import { CommonModule } from '@angular/common';
import {DynamicBrowseComponent} from "@components/dynamics/dynamic-browse/dynamic-browse.component";
import IView from "@interfaces/view.interfaces";
import {AuthService} from "@services/auth.service";
import {
  ContainerizationImagesModalCreateComponent
} from "@components/admin/containerization-images/containerization-images-modal-create/containerization-images-modal-create.component";
import {Router} from "@angular/router";
import {lastValueFrom} from "rxjs";
import {NotificationService, TypeNotificationEnum} from "@services/notification.service";
import {ModalInfoService} from "@services/modal-info.service";
import {ContainerizationImageItem} from "@models/containerization-images.model";
import {BackendService} from "@services/backend.service";
import {MessageLogService} from "@services/message-log.service";

@Component({
    selector: 'app-containerization-images-browser',
    imports: [CommonModule, DynamicBrowseComponent, ContainerizationImagesModalCreateComponent],
    templateUrl: './containerization-images-browser.component.html',
    styleUrls: ['./containerization-images-browser.component.sass']
})
export class ContainerizationImagesBrowserComponent implements IView {

  public BREADCRUMB_NAME = ''

  private readonly _modalInfo: ModalInfoService = inject(ModalInfoService)
  private readonly _notification: NotificationService = inject(NotificationService)
  private readonly _backend: BackendService = inject(BackendService)
  private readonly _messageLog = inject(MessageLogService)
  protected loading: boolean = false

  @ViewChild(ContainerizationImagesModalCreateComponent, {static: false}) containerizationImagesModalCreateComponent: ContainerizationImagesModalCreateComponent
  @ViewChild(DynamicBrowseComponent) dynamicBrowser: DynamicBrowseComponent

  private readonly _router = inject(Router)

  protected readonly _authService = inject(AuthService)

  protected createEvent(): void {
    this.containerizationImagesModalCreateComponent.openModal().then()
  }

  protected createItemSuccess(): void {
    this.dynamicBrowser.changePage();
  }

  openItem(id: number) {
    this._router.navigate(['containerizationImagesDetail', id]).then()
  }

  protected deleteItem(item: ContainerizationImageItem) {
    this._modalInfo.showModalInfo(
      'PROCESSES_ADMIN.CONTAINERIZATION_IMAGES.REMOVE_MODAL.TITLE',
      'PROCESSES_ADMIN.CONTAINERIZATION_IMAGES.REMOVE_MODAL.CONTENT',
      { name: item.name },
      'question-circle',
      'twotone',
      this._modalInfo.colorWarningDefault,
      [
        {
          text: "PROCESSES_ADMIN.CONTAINERIZATION_IMAGES.REMOVE_MODAL.ACCEPT_BUTTON",
          type: 'primary',
          onClickFunc: async () => {
            this._modalInfo.isVisibleModalInfo = false;
            this.loading = true;
            try {
              const response: any = await lastValueFrom(this._backend.deleteContainerizationImage(item.id));
              this._messageLog.addIssues(response.issues)
              this.dynamicBrowser.dynamicTable.checkedItems.clear();
              this.dynamicBrowser.changePage();
            } catch (e) {
              this._messageLog.addResponseIssues(e)
              this._notification.createNotificationWithType(
                TypeNotificationEnum.error,
                'PROCESSES_ADMIN.CONTAINERIZATION_IMAGES.REMOVE_MODAL.NOTIFICATION_ERROR.TITLE',
                'PROCESSES_ADMIN.CONTAINERIZATION_IMAGES.REMOVE_MODAL.NOTIFICATION_ERROR.CONTENT',
                'bottomRight',
              )
            }
            this.loading = false;
          }
        },
        {
          text: "PROCESSES_ADMIN.COMPUTE_RESOURCES.REMOVE_MODAL.CANCEL_BUTTON",
          type: 'primary',
          danger: true,
          onClickFunc: () => {
            this._modalInfo.isVisibleModalInfo = false;
          }
        },
      ]
    )
  }

}
