import { AfterViewChecked, ChangeDetectorRef, Component, ElementRef, OnDestroy, OnInit, ViewChild } from '@angular/core';
import { ActivatedRoute } from '@angular/router';
import { BackendService } from 'src/app/services/backend.service';
import { StateService } from 'src/app/services/state.service';
import { CommonModule } from '@angular/common';
import {SharedModule} from "../../../../shared-module/shared.module";
import {NzListModule} from "ng-zorro-antd/list";
import {NzInputModule} from "ng-zorro-antd/input";
import {FormsModule, ReactiveFormsModule} from "@angular/forms";
import {NzPageHeaderModule} from "ng-zorro-antd/page-header";
import {lastValueFrom} from "rxjs";

@Component({
    selector: 'app-system-functions-browse',
    imports: [
        CommonModule,
        SharedModule,
        NzListModule,
        NzInputModule,
        FormsModule,
        ReactiveFormsModule,
        NzPageHeaderModule,
    ],
    templateUrl: './system-functions-browse.component.html',
    styleUrls: ['./system-functions-browse.component.sass']
})
export class SystemFunctionsBrowseComponent implements OnInit, OnDestroy, AfterViewChecked {
  state: string;

  @ViewChild("containerPage", { static: false }) containerPage: ElementRef;

  list: any[];
  BREADCRUMB_NAME = 'SECURITY_ADMIN.SYSTEM_FUNCTIONS.BROWSER.BREADCRUMB'

  constructor(
    private activeRoute: ActivatedRoute,
    private stateService: StateService,
    private readonly backendService: BackendService,
    private readonly cdRef: ChangeDetectorRef,
  ) {
    this.state = '';
  }

  ngOnInit(): void {
    const SYSTEM_FUNCTIONS_DATA = this.activeRoute.snapshot.data.systemFunctions;
    this.list = SYSTEM_FUNCTIONS_DATA["content"];
    for (const f of this.list) {
      f['modified'] = false;
    }
  }

  ngAfterViewChecked() {
    this.cdRef.detectChanges();
  }

  getMaximumHeightList() {
    if (this.containerPage) {
      const containerDivPage = <HTMLDivElement>this.containerPage.nativeElement;
      var maximumHeight = containerDivPage.offsetHeight - 130;
      if (maximumHeight < 0) maximumHeight = 0;
      return `${maximumHeight}px`;
    }
    return '0px';
  }

  async saveExpression(systemFunction: {}) {
    await lastValueFrom(this.backendService.putACLExpression(systemFunction['acl_expression_id'],
      {'expression': systemFunction['can_execute_rule']}), { defaultValue: undefined });
    systemFunction['modified'] = false;
  }

  ruleChange(ev, systemFunction: {}) {
    systemFunction['can_execute_rule'] = ev;
    systemFunction['modified'] = true;//systemFunction['new_can_execute_rule'] !== systemFunction['can_execute_rule'];
  }

  ngOnDestroy(): void {

  }
}

  </nz-page-header>
