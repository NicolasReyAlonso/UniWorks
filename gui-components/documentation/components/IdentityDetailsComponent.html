import { Component, OnDestroy, OnInit, ViewChild } from '@angular/core';
import { ActivatedRoute, Router } from '@angular/router';
import { AuthService } from 'src/app/services/auth.service';
import { StateService } from '../../../../services/state.service';
import { IdentityFormComponent } from 'src/app/components/admin/identities/identity-form/identity-form.component';
import IView from "../../../../interfaces/view.interfaces";
import {CommonModule} from "@angular/common";
import {NzDescriptionsModule} from "ng-zorro-antd/descriptions";
import {NzPageHeaderModule} from "ng-zorro-antd/page-header";
import {BackButtonComponent} from "../../../../components/miscellaneous/back-button/back-button.component";
import {NzButtonModule} from "ng-zorro-antd/button";
import {SharedModule} from "../../../../shared-module/shared.module";

@Component({
    imports: [
        CommonModule,
        NzDescriptionsModule,
        NzPageHeaderModule,
        BackButtonComponent,
        NzButtonModule,
        SharedModule,
    ],
    selector: 'app-identity-details',
    templateUrl: './identity-details.component.html',
    styleUrls: ['./identity-details.component.sass']
})
export class IdentityDetailsComponent implements OnInit, OnDestroy, IView {

  BREADCRUMB_NAME = {
    key: 'SECURITY_ADMIN.IDENTITIES.DETAIL.BREADCRUMB',
    params: {
      name: '',
    }
  };

  state: string;

  identityData;
  groups;
  organizations;
  roles;
  identityId: string;

  @ViewChild("identityFormElementRef", { static: false }) identityFormElementRef: IdentityFormComponent;

  constructor(
    private readonly activeRoute: ActivatedRoute,
    private readonly stateService: StateService,
    private readonly router: Router,
    public readonly authService: AuthService,
  ) {
    this.state = '';
  }

  ngOnInit(): void {
    this.identityData = this.activeRoute.snapshot.data.identity["content"];
    this.groups = this.activeRoute.snapshot.data.groups["content"];
    this.organizations = this.activeRoute.snapshot.data.organizations["content"];
    this.roles = this.activeRoute.snapshot.data.roles["content"];
    this.identityId = this.identityData.id;
  }

  ngOnDestroy(): void {
    this.BREADCRUMB_NAME.params.name = this.identityData.name;
  }

  getGroupNameById(id) {
    for (const group of this.groups) {
      if (group.id == id) {
        return group.name;
      }
    }
  }

  getOrganizationNameById(id) {
    for (const organization of this.organizations) {
      if (organization.id == id) {
        return organization.name;
      }
    }
  }

  getRoleNameById(id) {
    for (const role of this.roles) {
      if (role.id == id) {
        return role.name;
      }
    }
  }

  getFormatDate(type: "creation_time" | "deactivation_time") {
    if (this.identityData && this.identityData[type]) {
      return this.identityData[type].replace("T", " ");
    } else {
      return "";
    }
  }

  goIdentityModify() {
    this.router.navigate(['identityModify', this.identityData.id]);
  }

}
