import {Component, OnDestroy, OnInit} from '@angular/core';
import {ActivatedRoute, Router} from '@angular/router';
import {forkJoin, lastValueFrom, Subscription} from 'rxjs';
import {AuthService} from 'src/app/services/auth.service';

import {BackendService} from 'src/app/services/backend.service';
import {MessageLogService} from 'src/app/services/message-log.service';
import {SocketService} from "src/app/services/socket.service";
import IView from "@interfaces/view.interfaces";
import { CommonModule } from '@angular/common';
import {NzListModule} from "ng-zorro-antd/list";
import {NzInputModule} from "ng-zorro-antd/input";
import {NzTableModule} from "ng-zorro-antd/table";
import {NzButtonModule} from "ng-zorro-antd/button";
import {FormsModule, ReactiveFormsModule} from "@angular/forms";
import {NzPageHeaderModule} from "ng-zorro-antd/page-header";
import {SharedModule} from "../../../shared-module/shared.module";
import {NzIconModule} from "ng-zorro-antd/icon";
import {ModalInfoService} from "@services/modal-info.service";

@Component({
    selector: 'app-processes-browse',
    imports: [
        CommonModule,
        NzListModule,
        NzInputModule,
        NzTableModule,
        NzButtonModule,
        FormsModule,
        ReactiveFormsModule,
        NzPageHeaderModule,
        SharedModule,
        NzIconModule,
    ],
    templateUrl: './processes-browse.component.html',
    styleUrls: ['./processes-browse.component.sass']
})
export class ProcessesBrowseComponent implements OnInit, OnDestroy, IView {

  BREADCRUMB_NAME = 'PROCESSES.BROWSER.BREADCRUMB.';

  constructor(
    private backendService: BackendService,
    private msg: MessageLogService,
    private socketService: SocketService,
    private route: ActivatedRoute,
    private readonly router: Router,
    public readonly authService: AuthService,
    private readonly modalInfoService: ModalInfoService,
  ) {
    const status = this.route.snapshot.queryParams.status;
    switch (status) {
      case 'todo':
        this.BREADCRUMB_NAME += 'TO_DO';
        break;
      case 'done':
        this.BREADCRUMB_NAME += 'DONE';
        break;
    }
  }

  title = 'PROCESSES.BROWSER.TITLE.';
  pageSize = 10;
  pageIndex = 1;
  totalItems = 0;
  status: string;
  jobs = [];
  loading = true;
  processStatusChangeSubscription: Subscription;
  computeResources = new Map<number, any>();
  processes = new Map<number, any>();
  loadDataSubscription: Subscription;

  async ngOnInit(): Promise<void> {
    this.setStatus();
    this.initProccessStatusChangeSubscription();
    await this.loadRelationship();
    this.loadData();
  }

  setStatus(): void {
    this.status = this.route.snapshot.queryParams.status;
    switch (this.status) {
      case 'todo':
        this.title += 'TO_DO';
        break;
      case 'done':
        this.title += 'DONE';
        break;
    }
  }

  ngOnDestroy(): void {
    if (this.processStatusChangeSubscription) {
      this.processStatusChangeSubscription.unsubscribe();
    }
  }

  private initProccessStatusChangeSubscription() {
    switch (this.status) {
      case 'todo':
        this.processStatusChangeSubscription = this.socketService.processStatusChangeSubject.subscribe({next:(event) => {
          for (const job of this.jobs) {
            if (job.id == event.id) {
              if (event.status == "success" || event.status == "error" || event.status == "cancel" || event.status == "cancelled") {
                this.loadData();
              } else {
                job.status = event.status;
              }
              break;
            }
          }
        }});
        break;
      case 'done':
        this.processStatusChangeSubscription = this.socketService.processStatusChangeSubject.subscribe({next:(job) => {
          if (job.status == "success" || job.status == "error" || job.status == "cancel" || job.status == "cancelled") {
            this.loadData();
          }
        }});
        break;
    }
  }

  async loadRelationship() {
    try {
      const response: any = await lastValueFrom(forkJoin({
        computeResource: this.backendService.getResources(),
        processes: this.backendService.getProcesses(),
      }));
      this.msg.addResponseIssues(response.computeResource);
      this.msg.addResponseIssues(response.processes);
      response.computeResource.content.forEach( (element) => {
        this.computeResources.set(element.id, element);
      });
      response.processes.content.forEach( (element) => {
        this.processes.set(element.id, element);
      });
    } catch (e) {
      this.msg.addResponseIssues(e);
      this.modalInfoService.showModalInfoDefaultError(e);
    }
  }

  loadData() {
    this.loading = false;
    if (this.loadDataSubscription) this.loadDataSubscription.unsubscribe();
    this.loadDataSubscription = this.backendService.getJobs(null, {
      status: this.status,
      pagination: {
        pageIndex: this.pageIndex,
        pageSize: this.pageSize,
      },
      order: {
        field: 'id',
        order: 'asc',
      }
    }).subscribe({
      next: (response: any) => {
        this.msg.addResponseIssues(response);
        this.jobs = response.content.map((element) => {
          return {
            ...element,
            processName: this.processes.get(element.process_id)?.name,
            resourceName: this.computeResources.get(element.resource_id)?.name,
          }
        });
        this.totalItems = response.count;
        this.loading = false;
      },
      error: (e) => {
        this.msg.addResponseIssues(e);
        this.modalInfoService.showModalInfoDefaultError(e);
        this.loading = false;
      }
    });
  }

  changePage(value) {
    this.pageIndex = value;
    this.loadData();
  }

  openJob(id) {
    this.router.navigate(['process', 'processDetail', id]);
  }

  // ITEM OPTIONS
  open(index) {
    this.msg.info('DUMMY: Abriendo el proceso número ' + index);
    console.log('DUMMY: Abriendo el proceso número ' + index);
  }

  export(index) {
    this.msg.info('DUMMY: Exportando el proceso número ' + index);
    console.log('DUMMY: Exportando el proceso número ' + index);
  }

  delete(index) {
    this.msg.info('DUMMY: Cancelando el proceso número ' + index);
    console.log('DUMMY: Cancelando el proceso número ' + index);
  }

}
